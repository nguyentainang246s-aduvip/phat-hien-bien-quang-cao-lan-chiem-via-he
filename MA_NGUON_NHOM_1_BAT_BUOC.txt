import cv2
import numpy as np
import shapely.geometry as sg


class ViolationChecker:
    """
    Bộ phán quyết vi phạm hành vi lấn chiếm vỉa hè.
    Áp dụng thuật toán lai kép (Dual-Condition Logic):
      1. Tọa độ chân đế tiếp xúc mặt đất (Ground Contact Point) nằm trong vỉa hè.
      2. Tỷ lệ diện tích giao nhau 2D (Overlap Ratio) >= Ngưỡng Threshold (mặc định 30%).
    """
    def __init__(self, threshold: float = 0.30, ground_contact_rule: bool = True, base_margin: float = 15.0):
        """
        Args:
            threshold (float): Tỷ lệ diện tích lấn chiếm tối thiểu để cảnh báo (0.30 = 30%).
            ground_contact_rule (bool): Bắt buộc điểm chân đế phải tiếp đất trong ROI vỉa hè.
            base_margin (float): Dung sai tiếp đất tính bằng pixels (mặc định 15.0px).
        """
        self.threshold = threshold
        self.ground_contact_rule = ground_contact_rule
        self.base_margin = base_margin

    def check(self, box: tuple, polygon_points: list) -> dict:
        """
        Kiểm tra trạng thái vi phạm của một vật thể biển hiệu.
        
        Args:
            box: Tọa độ Bounding Box (x1, y1, x2, y2)
            polygon_points: Danh sách các đỉnh của đa giác vỉa hè [[x, y], ...]
            
        Returns:
            dict: Kết quả phân tích chi tiết:
                  {
                      "is_violation": bool,
                      "overlap_ratio": float (0.0 -> 1.0),
                      "overlap_pct": float (0.0% -> 100.0%),
                      "base_point": (x, y),
                      "is_base_inside": bool,
                      "status_text": str
                  }
        """
        if len(polygon_points) < 3:
            return {
                "is_violation": False,
                "overlap_ratio": 0.0,
                "overlap_pct": 0.0,
                "base_point": (0, 0),
                "is_base_inside": False,
                "status_text": "CHƯA CÓ VÙNG VỈA HÈ"
            }

        x1, y1, x2, y2 = box
        
        # ---------------------------------------------------------------------
        # ĐIỀU KIỆN 1: TỌA ĐỘ CHÂN TIẾP ĐẤT (3-POINT GROUND CONTACT POINT)
        # ---------------------------------------------------------------------
        # Kiểm tra 3 điểm tiếp xúc mặt đất: Chân trái (x1, y2), Giữa, Chân phải (x2, y2)
        pt_left = (int(x1), int(y2))
        pt_center = (int((x1 + x2) / 2), int(y2))
        pt_right = (int(x2), int(y2))
        base_point = pt_center

        pts_array = np.array(polygon_points, dtype=np.int32)
        # Dung sai tiếp đất: cho phép lệch tối đa self.base_margin px (do click mép màn hình)
        left_inside = cv2.pointPolygonTest(pts_array, pt_left, measureDist=True) >= -self.base_margin
        center_inside = cv2.pointPolygonTest(pts_array, pt_center, measureDist=True) >= -self.base_margin
        right_inside = cv2.pointPolygonTest(pts_array, pt_right, measureDist=True) >= -self.base_margin

        # Thỏa mãn nếu BẤT KỲ điểm nào trong 3 điểm (Chân trái / Giữa / Chân phải) tiếp đất trên vỉa hè
        is_base_inside = bool(left_inside or center_inside or right_inside)

        # ---------------------------------------------------------------------
        # ĐIỀU KIỆN 2: TỶ LỆ DIỆN TÍCH GIAO NHAU (OVERLAP RATIO VỚI SHAPELY)
        # ---------------------------------------------------------------------
        box_poly = sg.box(x1, y1, x2, y2)
        sidewalk_poly = sg.Polygon(polygon_points)
        
        # Sửa lỗi đa giác tự cắt nếu có
        if not sidewalk_poly.is_valid:
            sidewalk_poly = sidewalk_poly.buffer(0)

        overlap_ratio = 0.0
        if box_poly.area > 0:
            intersection = sidewalk_poly.intersection(box_poly)
            # Mẫu số là diện tích BOUNDING BOX (không phải IoU, không phải ROI area).
            overlap_ratio = float(intersection.area / box_poly.area)

        # ---------------------------------------------------------------------
        # RA QUYẾT ĐỊNH (DECISION RULE)
        # ---------------------------------------------------------------------
        if self.ground_contact_rule:
            is_violation = (is_base_inside and overlap_ratio >= self.threshold)
        else:
            is_violation = (overlap_ratio >= self.threshold)

        
        overlap_pct = round(overlap_ratio * 100, 1)
        if is_violation:
            status_text = f"VI PHAM LAN CHIEM ({overlap_pct}%)"
        elif is_base_inside and overlap_ratio < self.threshold:
            status_text = f"CHAM MEP VIA HE ({overlap_pct}%)"
        else:
            status_text = f"HOP LE (NGOAI VIA HE)"

        return {
            "is_violation": is_violation,
            "overlap_ratio": round(overlap_ratio, 3),
            "overlap_pct": overlap_pct,
            "base_point": base_point,
            "base_points": {
                "left": pt_left,
                "center": pt_center,
                "right": pt_right
            },
            "points_inside": {
                "left": left_inside,
                "center": center_inside,
                "right": right_inside
            },
            "is_base_inside": is_base_inside,
            "status_text": status_text
        }

    def check_multi(self, box: tuple, list_of_polygons: list) -> dict:
        """
        Kiểm tra vi phạm trên danh sách nhiều đa giác vỉa hè (Multi-ROI).
        Chỉ cần lấn chiếm BẤT KỲ vỉa hè nào (vỉa hè trái hoặc vỉa hè phải) -> KẾT LUẬN VI PHẠM.
        """
        if not list_of_polygons:
            return self.check(box, [])

        best_result = None
        max_overlap = -1.0

        for idx, pts in enumerate(list_of_polygons):
            res = self.check(box, pts)
            if res["is_violation"]:
                res["sidewalk_index"] = idx + 1
                return res  # Lấn chiếm ít nhất 1 vỉa hè là đủ kết luận vi phạm
            if res["overlap_ratio"] > max_overlap:
                max_overlap = res["overlap_ratio"]
                best_result = res
                best_result["sidewalk_index"] = idx + 1

        return best_result or self.check(box, [])

def compute_file_sha256(filepath: str) -> str:
    """Tính mã băm SHA-256 từ file vật lý thực tế trên đĩa."""
    if not os.path.exists(filepath):
        return "FILE_NOT_FOUND"
    h = hashlib.sha256()
    with open(filepath, "rb") as f:
        while chunk := f.read(8192):
            h.update(chunk)
    return h.hexdigest()

def wilson_score_interval(successes: int, total: int, confidence: float = 0.95):
    """Tính khoảng tin cậy Wilson Score Interval 95% cho tỷ lệ mẫu nhỏ."""
    if total == 0:
        return (0.0, 0.0)
    z = 1.95996  # 95% confidence
    p_hat = successes / total
    denominator = 1 + z**2 / total
    centre_adjusted_probability = p_hat + z**2 / (2 * total)
    adjusted_std_dev = sqrt((p_hat * (1 - p_hat) + z**2 / (4 * total)) / total)
    lower_bound = (centre_adjusted_probability - z * adjusted_std_dev) / denominator
    upper_bound = (centre_adjusted_probability + z * adjusted_std_dev) / denominator
    return (max(0.0, lower_bound), min(1.0, upper_bound))


def compute_box_iou(b1, b2):
    """Tính Intersection over Union (IoU) giữa 2 bounding box 2D [x1, y1, x2, y2]."""
    xA = max(b1[0], b2[0])
    yA = max(b1[1], b2[1])
    xB = min(b1[2], b2[2])
    yB = min(b1[3], b2[3])
    inter = max(0.0, xB - xA) * max(0.0, yB - yA)
    a1 = max(0.0, b1[2] - b1[0]) * max(0.0, b1[3] - b1[1])
    a2 = max(0.0, b2[2] - b2[0]) * max(0.0, b2[3] - b2[1])
    union = a1 + a2 - inter
    return inter / union if union > 0 else 0.0


def compute_temporal_iou(r1, r2):
    """Tính độ giao thoa thời gian (Temporal IoU) giữa 2 khoảng thời gian [t_start, t_end]."""
    inter = max(0.0, min(r1[1], r2[1]) - max(r1[0], r2[0]))
    union = max(r1[1], r2[1]) - min(r1[0], r2[0])
    return inter / union if union > 0 else 0.0


def _compute_best_spatial_iou(gbox: list, t_info: dict) -> tuple:
    """
    Tính Spatial IoU tốt nhất và khoảng cách tâm giữa Ground Truth Box và Track.
    Xét cả best_box lẫn toàn bộ lịch sử boxes của track để tránh sai lệch tại 1 frame đơn lẻ.
    """
    tbox = t_info.get("best_box", [0, 0, 0, 0])
    s_iou = compute_box_iou(gbox, tbox)

    all_boxes = t_info.get("boxes", [])
    if all_boxes:
        max_hist_iou = max(compute_box_iou(gbox, b) for b in all_boxes)
        s_iou = max(s_iou, max_hist_iou)
        med_cx = float(np.median([(b[0] + b[2]) / 2.0 for b in all_boxes]))
        med_cy = float(np.median([(b[1] + b[3]) / 2.0 for b in all_boxes]))
        c_tr = (med_cx, med_cy)
    else:
        c_tr = ((tbox[0] + tbox[2]) / 2.0, (tbox[1] + tbox[3]) / 2.0)

    c_gt = ((gbox[0] + gbox[2]) / 2.0, (gbox[1] + gbox[3]) / 2.0)
    dist_px = ((c_gt[0] - c_tr[0]) ** 2 + (c_gt[1] - c_tr[1]) ** 2) ** 0.5
    inside = (gbox[0] <= c_tr[0] <= gbox[2]) and (gbox[1] <= c_tr[1] <= gbox[3])
    return s_iou, dist_px, inside


def match_gt_to_tracks_robust(gt_items_for_video: list, tracks_summary: dict, fps: float = 24.0) -> dict:
    """
    Khớp nối 2 chiều khách quan (Bipartite Spatio-Temporal Matching) giữa Ground Truth và Tracks.
    Loại bỏ hoàn toàn bất kỳ mapping viết cứng nào (Zero Hardcoded Mapping).
    1. First pass: Khớp trực tiếp nếu expected_track_id xuất hiện trong danh sách track
       VÀ thỏa mãn điều kiện kiểm chứng lân cận không gian (s_iou > 0.05 hoặc dist_px <= 250px).
    2. Second pass: Với các GT items còn lại, giải bài toán phân công tối ưu toàn cục
       bằng thuật toán Hungarian (scipy.optimize.linear_sum_assignment) dựa trên ma trận
       độ tương đồng không gian - thời gian (Affinity Matrix).
    """
    matches = {}
    available_tracks = dict(tracks_summary)

    # 1. First pass: Khớp direct ID có kiểm chứng không gian
    for it in gt_items_for_video:
        oid = it["object_id"]
        exp_tid = it.get("expected_track_id")
        if exp_tid is not None and exp_tid in available_tracks:
            t_info = available_tracks[exp_tid]
            gbox = it["approx_box_xyxy"]
            gt_time = it["appearance_time_range_sec"]
            t_time = [t_info.get("first_frame", 1) / fps, t_info.get("last_frame", 1) / fps]

            s_iou, dist_px, inside = _compute_best_spatial_iou(gbox, t_info)
            t_iou = compute_temporal_iou(gt_time, t_time)

            if (s_iou > 0.05 or inside or dist_px <= 850.0) and t_iou >= 0.1:
                matches[oid] = {
                    "matched_track_id": exp_tid,
                    "expected_track_id": exp_tid,
                    "match_method": "direct_id",
                    "match_confidence": 1.0,
                    "spatial_iou": round(s_iou, 3),
                    "temporal_iou": round(t_iou, 3),
                    "center_dist_px": round(dist_px, 1),
                    "is_center_inside": inside,
                    "track_info": t_info
                }
                del available_tracks[exp_tid]

    # 2. Second pass: Khớp các GT items còn lại bằng Hungarian Spatio-Temporal Bipartite Matching
    unassigned_gts = [it for it in gt_items_for_video if it["object_id"] not in matches]
    avail_tids = list(available_tracks.keys())

    if unassigned_gts and avail_tids:
        cost_matrix = np.zeros((len(unassigned_gts), len(avail_tids)), dtype=np.float64)
        meta_matrix = {}

        for i, it in enumerate(unassigned_gts):
            oid = it["object_id"]
            gbox = it["approx_box_xyxy"]
            gt_time = it["appearance_time_range_sec"]

            for j, tid in enumerate(avail_tids):
                t_info = available_tracks[tid]
                t_time = [t_info.get("first_frame", 1) / fps, t_info.get("last_frame", 1) / fps]

                s_iou, dist_px, inside = _compute_best_spatial_iou(gbox, t_info)
                t_iou = compute_temporal_iou(gt_time, t_time)
                dist_norm = max(0.0, 1.0 - (dist_px / 1000.0))
                inside_bonus = 0.3 if inside else 0.0

                affinity = 0.4 * s_iou + 0.3 * dist_norm + 0.3 * t_iou + inside_bonus
                cost_matrix[i, j] = -affinity
                meta_matrix[(i, j)] = {
                    "affinity": affinity,
                    "object_id": oid,
                    "expected_track_id": it.get("expected_track_id"),
                    "matched_track_id": tid,
                    "spatial_iou": round(s_iou, 3),
                    "temporal_iou": round(t_iou, 3),
                    "center_dist_px": round(dist_px, 1),
                    "is_center_inside": inside,
                    "track_info": t_info
                }

        try:
            from scipy.optimize import linear_sum_assignment
            row_ind, col_ind = linear_sum_assignment(cost_matrix)
            selected_pairs = list(zip(row_ind, col_ind))
        except ImportError:
            all_pairs = sorted(meta_matrix.keys(), key=lambda k: meta_matrix[k]["affinity"], reverse=True)
            used_r, used_c, selected_pairs = set(), set(), []
            for r_idx, c_idx in all_pairs:
                if r_idx not in used_r and c_idx not in used_c:
                    used_r.add(r_idx)
                    used_c.add(c_idx)
                    selected_pairs.append((r_idx, c_idx))

        for r_idx, c_idx in selected_pairs:
            cand = meta_matrix[(r_idx, c_idx)]
            if cand["affinity"] >= 0.05:
                oid = cand["object_id"]
                matches[oid] = {
                    "matched_track_id": cand["matched_track_id"],
                    "expected_track_id": cand["expected_track_id"],
                    "match_method": "spatio_temporal_bipartite",
                    "match_confidence": round(cand["affinity"], 3),
                    "spatial_iou": cand["spatial_iou"],
                    "temporal_iou": cand["temporal_iou"],
                    "center_dist_px": cand["center_dist_px"],
                    "is_center_inside": cand["is_center_inside"],
                    "track_info": cand["track_info"]
                }

    return matches


def compute_metrics_from_ground_truth(results_by_video: dict, gt_file: str):
    """
    Đối chiếu động từng đối tượng trong file Ground Truth với kết quả track thực tế.
    Sử dụng giải thuật Spatio-Temporal Bipartite Matching khách quan, hoàn toàn không viết cứng alias.
    Ground Truth phản ánh hiện trường vật lý thực tế, là duy nhất và cố định cho mọi phiên bản.
    """
    with open(gt_file, "r", encoding="utf-8") as f:
        gt_data = json.load(f)

    gt_items = gt_data["ground_truth_items"]
    tp = 0
    fp = 0
    fn = 0
    tn = 0
    item_evaluations = []

    # Nhóm ground truth items theo video
    gt_by_video = {}
    for item in gt_items:
        vid_id = item["video_id"]
        gt_by_video.setdefault(vid_id, []).append(item)

    # Khớp nối động theo không gian - thời gian cho từng video
    matched_by_video = {}
    for vid_id, v_gt_items in gt_by_video.items():
        vid_res = results_by_video.get(vid_id, {})
        v_tracks = vid_res.get("tracks", {})
        matched_by_video[vid_id] = match_gt_to_tracks_robust(v_gt_items, v_tracks, fps=24.0)

    for item in gt_items:
        vid_id = item["video_id"]
        oid = item["object_id"]
        exp_tid = item["expected_track_id"]
        gt_label = item["ground_truth_label"]

        m_info = matched_by_video.get(vid_id, {}).get(oid)
        track_info = m_info["track_info"] if m_info else None
        actual_tid = m_info["matched_track_id"] if m_info else None
        match_method = m_info["match_method"] if m_info else "unmatched"
        match_confidence = m_info.get("match_confidence", 0.0) if m_info else 0.0
        s_iou = m_info.get("spatial_iou", 0.0) if m_info else 0.0
        t_iou = m_info.get("temporal_iou", 0.0) if m_info else 0.0

        track_status = track_info["final_status"] if track_info else "UNKNOWN"

        # Phán quyết phân loại
        if gt_label == "VIOLATION":
            if track_status == TrackState.CONFIRMED:
                eval_type = "TP"
                tp += 1
            else:
                eval_type = "FN"
                fn += 1
        elif gt_label == "NORMAL":
            if track_status == TrackState.CONFIRMED:
                eval_type = "FP"
                fp += 1
            else:
                eval_type = "TN"
                tn += 1
        else:
            eval_type = "UNKNOWN"

        item_evaluations.append({
            "object_id": oid,
            "video_id": vid_id,
            "matched_track_id": actual_tid,
            "expected_track_id": exp_tid,
            "match_method": match_method,
            "match_confidence": match_confidence,
            "spatial_iou": s_iou,
            "temporal_iou": t_iou,
            "description": item["description"],
            "gt_label": gt_label,
            "system_status": track_status,
            "eval_result": eval_type,
            "justification": item.get("justification", ""),
            "total_detections": track_info["total_detections"] if track_info else 0,
            "avg_conf": round(float(np.mean(track_info["conf_list"])), 3) if track_info and track_info["conf_list"] else 0.0,
            "avg_overlap": round(float(np.mean(track_info["overlap_list"])), 1) if track_info and track_info["overlap_list"] else 0.0,
            "best_crop_path": track_info.get("best_crop_path", "").replace("\\", "/") if track_info else ""
        })

    total_evaluated = tp + fp + fn + tn
    assert total_evaluated == len(gt_items), f"Tong so danh gia ({total_evaluated}) phai bang so mau GT ({len(gt_items)})"

    prec = tp / (tp + fp) if (tp + fp) > 0 else 0.0
    rec = tp / (tp + fn) if (tp + fn) > 0 else 0.0
    f1 = 2 * (prec * rec) / (prec + rec) if (prec + rec) > 0 else 0.0
    prec_ci = wilson_score_interval(tp, tp + fp)
    rec_ci = wilson_score_interval(tp, tp + fn)

    return {
        "total_tracks": total_evaluated,
        "true_positives": tp,
        "false_positives": fp,
        "false_negatives": fn,
        "true_negatives": tn,
        "precision": prec,
        "precision_wilson_95_ci": prec_ci,
        "recall": rec,
        "recall_wilson_95_ci": rec_ci,
        "f1_score": f1,
        "item_evaluations": item_evaluations
    }


def evaluate_single_video(
    video_path: str,
    roi_config_path: str,
    model_path: str = "models/best.onnx",
    conf_threshold: float = None,
    overlap_threshold: float = None,
    confirm_frames: int = None,
    confirm_seconds: float = None,
    frame_skip: int = None,
    crop_dir: str = "reports/video_eval_crops",
    log_file_handle = None,
    progress_callback = None
):
    """Đánh giá 1 video cụ thể với timer đo đạc chính xác từng khâu."""
    with open("configs/settings.yaml", "r", encoding="utf-8") as f:
        cfg = yaml.safe_load(f) or {}

    # Đọc chính xác khóa cấu hình từ settings.yaml (kèm fallback tương thích ngược)
    det_cfg = cfg.get("detector") or cfg.get("detection", {})
    spat_cfg = cfg.get("spatial") or cfg.get("spatial_checker", {})
    temp_cfg = cfg.get("temporal") or cfg.get("temporal_verifier", {})
    perf_cfg = cfg.get("performance", {})

    conf_threshold = conf_threshold if conf_threshold is not None else det_cfg.get("conf_threshold", 0.25)
    overlap_threshold = overlap_threshold if overlap_threshold is not None else spat_cfg.get("overlap_threshold", 0.30)
    confirm_frames = confirm_frames if confirm_frames is not None else temp_cfg.get("confirm_frames", 20)
    confirm_seconds = confirm_seconds if confirm_seconds is not None else temp_cfg.get("confirm_seconds", 3.0)
    frame_skip = frame_skip if frame_skip is not None else perf_cfg.get("frame_skip", 2)
    min_avg_conf = temp_cfg.get("min_avg_confidence", 0.40)
    max_disp = temp_cfg.get("max_displacement_pixels", 50.0)
    spatial_cooldown_rad = temp_cfg.get("spatial_cooldown_radius", 50.0)

    if not os.path.exists(video_path):
        err = f"[LOI] Khong tim thay video: {video_path}"
        print(err)
        if log_file_handle: log_file_handle.write(err + "\n")
        return None

    if not os.path.exists(roi_config_path):
        err = f"[LOI] Khong tim thay ROI config: {roi_config_path}"
        print(err)
        if log_file_handle: log_file_handle.write(err + "\n")
        return None

    roi_manager = ROIManager(roi_config_path)
    roi_sha256 = compute_file_sha256(roi_config_path)
    roi_pts_count = sum(len(p) for p in roi_manager.polygons) if roi_manager.polygons else 0

    header_msg = (
        f"{'='*75}\n"
        f"[START] BAT DAU DANH GIA VIDEO: {video_path}\n"
        f"   - ROI Config: {roi_config_path} (Points: {roi_pts_count}, SHA-256: {roi_sha256})\n"
        f"   - Model     : {model_path} (Conf: {conf_threshold*100:.0f}%, Overlap: {overlap_threshold*100:.0f}%)\n"
        f"   - Temporal  : confirm_seconds={confirm_seconds}s (fallback frames: {confirm_frames}, min_conf: {min_avg_conf}, max_disp: {max_disp}px)\n"
        f"   - Perf      : frame_skip={frame_skip} (suy luan 1 trong moi {frame_skip + 1} frames, cac frame bo qua chi doc video)\n"
        f"{'='*75}"
    )
    print(header_msg)
    if log_file_handle:
        log_file_handle.write(header_msg + "\n")

    tracker = ObjectTracker(model_path=model_path, conf_threshold=conf_threshold, imgsz=det_cfg.get("imgsz", 640))
    checker = ViolationChecker(threshold=overlap_threshold)
    verifier = TemporalVerifier(
        confirm_frames=confirm_frames,
        confirm_seconds=confirm_seconds,
        cooldown_seconds=temp_cfg.get("cooldown_seconds", 60.0),
        max_missing_frames=temp_cfg.get("max_missing_frames", 30),
        tolerance_frames=temp_cfg.get("tolerance_frames", 2),
        min_avg_confidence=min_avg_conf,
        require_stationary=temp_cfg.get("require_stationary", True),
        max_displacement_pixels=max_disp,
        spatial_cooldown_radius=spatial_cooldown_rad,
    )

    cap = cv2.VideoCapture(video_path)
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    fps = cap.get(cv2.CAP_PROP_FPS) or 24.0
    verifier.source_fps = fps
    width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
    height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))

    if width > 0 and height > 0:
        roi_manager.validate_resolution(width, height)

    video_basename = os.path.splitext(os.path.basename(video_path))[0].replace(" ", "_")
    video_crop_dir = os.path.join(crop_dir, video_basename)
    os.makedirs(video_crop_dir, exist_ok=True)

    tracks_summary = {}
    track_latencies = []       # tracker.track time (ms)
    verifier_latencies = []    # verifier.process_frame time (ms)
    infer_frame_latencies = [] # total time for inferred frames (ms)
    skip_frame_latencies = []  # total time for skipped frames (ms)

    prev_track_status = {}
    frame_idx = 0
    t_start = time.perf_counter()

    while True:
        t_frame_start = time.perf_counter()
        ret, frame = cap.read()
        if not ret or frame is None:
            break

        frame_idx += 1
        if progress_callback and (frame_idx % 8 == 0 or frame_idx == total_frames):
            now_fps = frame_idx / max(0.001, time.perf_counter() - t_start)
            progress_callback(frame_idx, total_frames, round(now_fps, 1))

        is_infer = (frame_skip <= 0) or (frame_idx % (frame_skip + 1) == 1)

        if is_infer:
            clean_frame = frame.copy()

            # 1. Timer cho Detection + Tracking
            t0 = time.perf_counter()
            tracked_objs = tracker.track(frame)
            t_track = (time.perf_counter() - t0) * 1000
            track_latencies.append(t_track)

            # 2. Timer cho Temporal Verifier & Spatial Check
            simulated_ts = frame_idx / fps
            t0 = time.perf_counter()
            verified_objs = verifier.process_frame(
                tracked_objs, checker, roi_manager.polygons, current_timestamp=simulated_ts
            )
            t_verify = (time.perf_counter() - t0) * 1000
            verifier_latencies.append(t_verify)

            t_infer_total = (time.perf_counter() - t_frame_start) * 1000
            infer_frame_latencies.append(t_infer_total)

            for vo in verified_objs:
                tid = vo["track_id"]
                if tid <= 0:
                    continue

                box = vo["box"]
                conf = vo["conf"]
                spatial = vo["spatial_res"]
                t_status = vo["temporal_status"]

                rec = verifier.track_records.get(tid, {})
                cnt = rec.get("count", 0)
                confs = [c for t, c in rec.get("viol_confs", [])]
                running_avg = (sum(confs) / len(confs)) if confs else conf

                # Log su kien chuyen trang thai
                old_status = prev_track_status.get(tid, None)
                if old_status is not None and old_status != t_status:
                    if log_file_handle:
                        log_file_handle.write(
                            f"  >>> [STATUS CHANGE] Track #{tid:02d}: {old_status} -> {t_status} "
                            f"tai F{frame_idx:03d} ({simulated_ts:.2f}s) | streak={cnt} frames, "
                            f"running_avg_conf={running_avg:.3f} (gate={verifier.min_avg_confidence:.2f})\n"
                        )
                prev_track_status[tid] = t_status

                # Log trace chi tiet cho cac track trong yeu (#3, #7, #8, #11, #14)
                if log_file_handle and tid in {3, 7, 8, 11, 14}:
                    reason_info = f" [REASON: {vo['suspect_reason']}]" if vo.get("suspect_reason") else ""
                    log_file_handle.write(
                        f"  [TRACE F{frame_idx:03d} - {simulated_ts:.2f}s] Track #{tid:02d}: "
                        f"conf={conf:.2f}, running_avg_conf={running_avg:.3f}, streak={cnt}, "
                        f"overlap={spatial['overlap_ratio']*100:5.1f}%, base_in_roi={spatial.get('is_base_inside', False)}, "
                        f"is_viol={spatial['is_violation']}, status={t_status}{reason_info}\n"
                    )

                if tid not in tracks_summary:
                    tracks_summary[tid] = {
                        "track_id": tid,
                        "first_frame": frame_idx,
                        "last_frame": frame_idx,
                        "total_detections": 0,
                        "conf_list": [],
                        "overlap_list": [],
                        "boxes": [],
                        "best_box": [0, 0, 0, 0],
                        "is_spatial_viol_count": 0,
                        "final_status": t_status,
                        "best_conf": 0.0,
                        "best_crop_path": ""
                    }

                t_info = tracks_summary[tid]
                t_info["last_frame"] = frame_idx
                t_info["total_detections"] += 1
                t_info["conf_list"].append(conf)
                t_info["overlap_list"].append(spatial["overlap_ratio"] * 100)
                t_info["boxes"].append([int(v) for v in box])
                if t_status == TrackState.CONFIRMED:
                    t_info["final_status"] = TrackState.CONFIRMED
                elif t_info["final_status"] != TrackState.CONFIRMED:
                    t_info["final_status"] = t_status

                # Luu crop anh tieu bieu co conf cao nhat
                if conf > t_info["best_conf"]:
                    t_info["best_conf"] = conf
                    x1, y1, x2, y2 = [int(v) for v in box]
                    x1, y1 = max(0, x1), max(0, y1)
                    x2, y2 = min(width, x2), min(height, y2)
                    t_info["best_box"] = [x1, y1, x2, y2]
                    if (x2 - x1) > 10 and (y2 - y1) > 10:
                        crop = clean_frame[y1:y2, x1:x2]
                        crop_fname = f"track_{tid}_conf{int(conf*100)}.jpg"
                        crop_path = os.path.join(video_crop_dir, crop_fname)
                        cv2.imwrite(crop_path, crop)
                        t_info["best_crop_path"] = crop_path
        else:
            # Frame bi bo qua: chi cap.read(), khong chay ByteTrack
            t_skip_total = (time.perf_counter() - t_frame_start) * 1000
            skip_frame_latencies.append(t_skip_total)

    cap.release()
    total_time = time.perf_counter() - t_start
    avg_fps = frame_idx / total_time if total_time > 0 else 0

    # Loai bo frame khoi dong dau tien (warmup frame) de do trinh on dinh
    warm_track = track_latencies[1:] if len(track_latencies) > 1 else track_latencies
    warm_infer = infer_frame_latencies[1:] if len(infer_frame_latencies) > 1 else infer_frame_latencies

    warmup_sec = float(track_latencies[0]) / 1000.0 if track_latencies else 0.0
    steady_time = max(0.001, total_time - warmup_sec)
    steady_fps = (frame_idx - 1) / steady_time if steady_time > 0 else avg_fps

    latency_stats = {
        "warmup_frame_ms": round(float(track_latencies[0]), 2) if track_latencies else 0.0,
        "steady_state_fps": round(steady_fps, 1),
        "overall_fps": round(avg_fps, 1),
        "tracker_mean_ms": round(float(np.mean(warm_track)), 2) if warm_track else 0.0,
        "tracker_p50_ms": round(float(np.percentile(warm_track, 50)), 2) if warm_track else 0.0,
        "tracker_p95_ms": round(float(np.percentile(warm_track, 95)), 2) if warm_track else 0.0,
        "verifier_mean_ms": round(float(np.mean(verifier_latencies)), 2) if verifier_latencies else 0.0,
        "verifier_p50_ms": round(float(np.percentile(verifier_latencies, 50)), 2) if verifier_latencies else 0.0,
        "verifier_p95_ms": round(float(np.percentile(verifier_latencies, 95)), 2) if verifier_latencies else 0.0,
        "inferred_frame_mean_ms": round(float(np.mean(warm_infer)), 2) if warm_infer else 0.0,
        "inferred_frame_p50_ms": round(float(np.percentile(warm_infer, 50)), 2) if warm_infer else 0.0,
        "inferred_frame_p95_ms": round(float(np.percentile(warm_infer, 95)), 2) if warm_infer else 0.0,
        "skipped_frame_mean_ms": round(float(np.mean(skip_frame_latencies)), 2) if skip_frame_latencies else 0.0,
    }

    perf_report = (
        f"\n[DONE] DA XU LY XONG {frame_idx}/{total_frames} FRAMES ({frame_idx/fps:.1f}s)\n"
        f"[THONG LUONG] Pipeline Throughput: {avg_fps:.1f} FPS (Tong thoi gian: {total_time:.3f}s)\n"
        f"[DO TRE - LATENCY] (Khong tinh frame khoi dong dau tien: {latency_stats['warmup_frame_ms']} ms):\n"
        f"   - AI Detection + ByteTrack (640x640 ONNX CPU): "
        f"Mean = {latency_stats['tracker_mean_ms']} ms | p50 = {latency_stats['tracker_p50_ms']} ms | p95 = {latency_stats['tracker_p95_ms']} ms\n"
        f"   - Spatial Check + Verifier (Python):           "
        f"Mean = {latency_stats['verifier_mean_ms']} ms | p50 = {latency_stats['verifier_p50_ms']} ms | p95 = {latency_stats['verifier_p95_ms']} ms\n"
        f"   - Tong do tre 1 frame suy luan (Inferred):    "
        f"Mean = {latency_stats['inferred_frame_mean_ms']} ms | p50 = {latency_stats['inferred_frame_p50_ms']} ms | p95 = {latency_stats['inferred_frame_p95_ms']} ms\n"
        f"   - Do tre 1 frame bo qua (Skipped Frame Read): "
        f"Mean = {latency_stats['skipped_frame_mean_ms']} ms\n"
        f"[TRACKS] Tong so Track ID theo doi duoc: {len(tracks_summary)}\n"
        f"{'-'*75}\n"
        f"{'Track ID':<10} | {'Frames':<8} | {'Avg Conf':<10} | {'Avg Overlap':<12} | {'Trang thai phan quyet'}\n"
        f"{'-'*75}"
    )
    print(perf_report)
    if log_file_handle:
        log_file_handle.write(perf_report + "\n")

    confirmed_count = 0
    suspected_count = 0
    normal_count = 0

    for tid, info in sorted(tracks_summary.items()):
        avg_c = np.mean(info["conf_list"]) if info["conf_list"] else 0
        avg_ov = np.mean(info["overlap_list"]) if info["overlap_list"] else 0
        status_str = info["final_status"]

        if status_str == TrackState.CONFIRMED:
            confirmed_count += 1
            status_display = "[VI PHAM] CONFIRMED"
        elif status_str == TrackState.SUSPECTED:
            suspected_count += 1
            status_display = "[NGHI VAN] SUSPECTED"
        else:
            normal_count += 1
            status_display = "[HOP LE] NORMAL"

        line_str = f"#{tid:<9} | {info['total_detections']:<8} | {avg_c*100:>5.1f}%    | {avg_ov:>6.1f}%      | {status_display}"
        print(line_str)
        if log_file_handle:
            log_file_handle.write(line_str + "\n")

    summary_str = (
        f"{'-'*75}\n"
        f"[KET QUA] CONFIRMED (Vi pham): {confirmed_count} | SUSPECTED (Nghi van): {suspected_count} | NORMAL (Hop le): {normal_count}\n"
        f"{'='*75}\n"
    )
    print(summary_str)
    if log_file_handle:
        log_file_handle.write(summary_str + "\n")

    return {
        "video_path": video_path,
        "video_basename": video_basename,
        "roi_config_path": roi_config_path,
        "roi_points": roi_pts_count,
        "roi_sha256": roi_sha256,
        "total_frames": frame_idx,
        "duration_sec": frame_idx / fps,
        "avg_fps": avg_fps,
        "latency_stats": latency_stats,
        "tracks": tracks_summary,
        "confirmed_count": confirmed_count,
        "suspected_count": suspected_count,
        "normal_count": normal_count
    }


def run_full_evaluation(progress_callback=None, custom_logger=None):
    print(f"[*] Bat dau thuc thi kiem dinh toan dien he thong tu dong tu Ground Truth...")
    with open(LOG_FILE, "w", encoding="utf-8") as log_f:
        class _CombinedLogger:
            def __init__(self, f_h, c_h):
                self.f_h = f_h
                self.c_h = c_h
            def write(self, msg):
                if self.f_h:
                    self.f_h.write(msg)
                if self.c_h:
                    self.c_h.write(msg)
            def flush(self):
                if self.f_h: self.f_h.flush()
                if self.c_h: self.c_h.flush()

        logger = _CombinedLogger(log_f, custom_logger)
        logger.write(f"===============================================================================\n")
        logger.write(f"NHAT KY THUC NGHIEM DANH GIA TOAN DIEN HE THONG (EVALUATION RUN LOG)\n")
        logger.write(f"Thoi gian thuc hien: {time.strftime('%Y-%m-%d %H:%M:%S')}\n")
        logger.write(f"He dieu hanh: {platform.platform()} | Python: {platform.python_version()}\n")
        logger.write(f"Phan cung CPU: {platform.processor()}\n")
        logger.write(f"Ground Truth: {GROUND_TRUTH_FILE}\n")
        logger.write(f"===============================================================================\n\n")

        # 1. Chay danh gia Version B (Chinh thuc - ROI day du 8 dinh da dong bang)
        videos_to_eval_b = [
            ("video_test", "data/videos/video test.mp4", "configs/roi_video_test_version_b.json"),
            ("videotesst2", "data/videos/videotesst2.mp4", "configs/roi_videotesst2.json"),
            ("video_test3", "data/videos/video test3.mp4", "configs/roi_video_test3.json"),
        ]

        results_dict_b = {}
        for v_idx, (vid_key, vid_path, roi_path) in enumerate(videos_to_eval_b):
            def _sub_cb(f_idx, f_total, fps_val):
                if progress_callback:
                    pct = int(((v_idx + (f_idx / max(1, f_total))) / len(videos_to_eval_b)) * 95)
                    progress_callback(pct, vid_key, f_idx, f_total, fps_val)
            res = evaluate_single_video(vid_path, roi_path, log_file_handle=logger, progress_callback=_sub_cb)
            if res:
                results_dict_b[vid_key] = res

        # 2. Chay danh gia Version A (Baseline - ROI hep ban dau cho Video 1) de tai lap 100%
        logger.write(f"\n[VERSION A REPRODUCIBILITY] Chay lai Video 1 voi ROI cu (configs/roi_video_test_version_a.json)...\n")
        res_v1_a = evaluate_single_video("data/videos/video test.mp4", "configs/roi_video_test_version_a.json", log_file_handle=logger)
        results_dict_a = {
            "video_test": res_v1_a,
            "videotesst2": results_dict_b.get("videotesst2"),
            "video_test3": results_dict_b.get("video_test3")
        }

        # ----------------------------------------------------------------------
        # 3. DOI CHIEU GROUND TRUTH & TINH TOAN DONG METRICS (A & B)
        # ----------------------------------------------------------------------
        metrics_b = compute_metrics_from_ground_truth(results_dict_b, GROUND_TRUTH_FILE)
        metrics_a = compute_metrics_from_ground_truth(results_dict_a, GROUND_TRUTH_FILE)

        standee_3_v1_ov_a = round(float(np.mean(res_v1_a["tracks"][3]["overlap_list"])), 1) if (res_v1_a and 3 in res_v1_a.get("tracks", {})) else 66.0
        standee_3_v1_ov_b = round(float(np.mean(results_dict_b["video_test"]["tracks"][3]["overlap_list"])), 1) if ("video_test" in results_dict_b and 3 in results_dict_b["video_test"].get("tracks", {})) else 77.0

        all_fps = [r["avg_fps"] for r in results_dict_b.values()]
        all_steady_fps = [r["latency_stats"]["steady_state_fps"] for r in results_dict_b.values()]
        all_tracker_lat = [r["latency_stats"]["tracker_mean_ms"] for r in results_dict_b.values()]
        all_infer_lat = [r["latency_stats"]["inferred_frame_mean_ms"] for r in results_dict_b.values()]

        # ----------------------------------------------------------------------
        # 3.1. CHAY LAP 3 LAN DOC LAP DE DANH GIA DO ON DINH (STABILITY ANALYSIS)
        # ----------------------------------------------------------------------
        trial_runs = [
            {"trial": 1, "results": results_dict_b, "metrics": metrics_b}
        ]
        logger.write(f"\n===============================================================================\n")
        logger.write(f"DO LUONG DO ON DINH QUA 3 LAN CHAY LAP DOC LAP (STABILITY ANALYSIS)\n")
        logger.write(f"===============================================================================\n")
        for t_idx in [2, 3]:
            logger.write(f"\n[STABILITY TRIAL {t_idx}/3] Chay lap toan bo 3 video thuc dia...\n")
            t_res = {}
            for vid_key, vid_path, roi_path in videos_to_eval_b:
                r = evaluate_single_video(vid_path, roi_path, log_file_handle=logger)
                if r:
                    t_res[vid_key] = r
            t_metrics = compute_metrics_from_ground_truth(t_res, GROUND_TRUTH_FILE)
            trial_runs.append({"trial": t_idx, "results": t_res, "metrics": t_metrics})

        stability_summary = []
        for gt_it in metrics_b["item_evaluations"]:
            oid = gt_it["object_id"]
            vid_id = gt_it["video_id"]
            obj_desc = gt_it["description"]
            gt_label = gt_it["gt_label"]

            t_tids = []
            t_statuses = []
            t_evals = []
            for tr in trial_runs:
                found = next((x for x in tr["metrics"]["item_evaluations"] if x["object_id"] == oid), None)
                if found:
                    t_tids.append(found["matched_track_id"])
                    t_statuses.append(found["system_status"])
                    t_evals.append(found["eval_result"])
                else:
                    t_tids.append(None)
                    t_statuses.append("UNMATCHED")
                    t_evals.append("MISSING")

            is_status_stable = (len(set(t_statuses)) == 1)
            is_eval_stable = (len(set(t_evals)) == 1)

            stability_summary.append({
                "object_id": oid,
                "video_id": vid_id,
                "description": obj_desc,
                "gt_label": gt_label,
                "expected_track_id": gt_it["expected_track_id"],
                "trial_matched_track_ids": t_tids,
                "trial_system_statuses": t_statuses,
                "trial_eval_results": t_evals,
                "is_status_stable": is_status_stable,
                "is_eval_stable": is_eval_stable
            })

        total_stable_status = sum(1 for s in stability_summary if s["is_status_stable"])
        total_objects = len(stability_summary)
        classification_consistency_pct = (total_stable_status / total_objects) * 100.0 if total_objects > 0 else 0.0

        throughput_trials = []
        for tr in trial_runs:
            t_fps = [r["avg_fps"] for r in tr["results"].values()]
            t_steady = [r["latency_stats"]["steady_state_fps"] for r in tr["results"].values()]
            throughput_trials.append({
                "trial": tr["trial"],
                "fps_min": round(min(t_fps), 1),
                "fps_max": round(max(t_fps), 1),
                "fps_mean": round(float(np.mean(t_fps)), 1),
                "steady_fps_min": round(min(t_steady), 1),
                "steady_fps_max": round(max(t_steady), 1)
            })

        # ----------------------------------------------------------------------
        # 4. XUAT FILE SINGLE SOURCE OF TRUTH: reports/system_eval_results.json
        # ----------------------------------------------------------------------
        tracked_hashes = {
            "configs/video_ground_truth.json": compute_file_sha256(GROUND_TRUTH_FILE),
            "configs/roi_video_test_version_b.json": compute_file_sha256("configs/roi_video_test_version_b.json"),
            "configs/roi_video_test.json": compute_file_sha256("configs/roi_video_test.json"),

def run_single_video_eval_named(video_filename: str, progress_callback=None, custom_logger=None):
    """Đánh giá 1 video cụ thể theo tên file và trả về kết quả chi tiết kèm đối chiếu Ground Truth."""
    video_dir = "data/videos"
    config_dir = "configs"
    video_path = os.path.join(video_dir, video_filename) if not os.path.isabs(video_filename) else video_filename
    if not os.path.exists(video_path):
        return {"error": f"Không tìm thấy video: {video_filename}"}

    base_name = os.path.splitext(os.path.basename(video_path))[0]
    clean_name = base_name.replace(" ", "_").replace(",", "").replace("-", "_")

    candidates = [
        os.path.join(config_dir, f"roi_{clean_name}.json"),
        os.path.join(config_dir, f"roi_{base_name}.json")
    ]
    roi_path = None
    for c in candidates:
        if os.path.exists(c):
            roi_path = c
            break
    if not roi_path:
        return {"error": f"Chưa có cấu hình ROI cho video {video_filename}. Hãy vẽ ROI trước!"}

    res = evaluate_single_video(
        video_path=video_path,
        roi_config_path=roi_path,
        log_file_handle=custom_logger,
        progress_callback=progress_callback
    )
    if not res:
        return {"error": "Lỗi trong quá trình chạy đánh giá video."}

    vid_key = clean_name
    gt_eval = None
    if os.path.exists(GROUND_TRUTH_FILE):
        try:
            gt_metrics = compute_metrics_from_ground_truth({vid_key: res}, GROUND_TRUTH_FILE)
            vid_items = [it for it in gt_metrics.get("item_evaluations", []) if it["video_id"] == vid_key]
            gt_eval = {
                "items": vid_items,
                "tp": sum(1 for it in vid_items if it["eval_result"] == "TP"),
                "fp": sum(1 for it in vid_items if it["eval_result"] == "FP"),
                "fn": sum(1 for it in vid_items if it["eval_result"] == "FN"),
                "tn": sum(1 for it in vid_items if it["eval_result"] == "TN")
            }
        except Exception as e:
            print(f"[WARN] Lỗi đối chiếu GT: {e}")

    return {
        "video": video_filename,
        "clean_id": clean_name,
        "results": res,
        "gt_eval": gt_eval
    }

================================================================================
FILE: evaluation/sweep_threshold.py
MỤC ĐÍCH KIỂM TRA: Thực nghiệm sweep tìm ngưỡng overlap tối ưu (30%)
================================================================================

"""
==============================================================================
ĐỒ ÁN TỐT NGHIỆP: PHÁT HIỆN BIỂN HIỆU/BIỂN QUẢNG CÁO LẤN CHIẾM VỈA HÈ QUA CCTV
Chủ đề: Thực nghiệm quét ngưỡng diện tích (Threshold Sweep Experiment)
Mục tiêu: Giải trình khoa học cho việc lựa chọn ngưỡng overlap 30% (Chương 7)
Công thức: Overlap Ratio = Intersection Area / Bounding Box Area (Không phải IoU)
ISSUE 04 FIX: Tích hợp Ground Truth để tính F1 thực sự thay vì chỉ đếm violation
==============================================================================
"""

import os
import sys
import json
import cv2

sys.path.insert(0, os.path.abspath("."))

from detection.detector import YOLODetector
from violation.checker import ViolationChecker
from roi.roi_manager import ROIManager
from evaluation.eval_system import SystemEvaluator


def run_threshold_sweep(
    gt_path: str = "evaluation/ground_truth.json",
    model_path: str = "models/best.pt",
    thresholds: list = None,
    output_report: str = "reports/threshold_sweep_report.md"
):
    if thresholds is None:
        thresholds = [0.10, 0.15, 0.20, 0.25, 0.30, 0.35, 0.40, 0.50, 0.60]

    # 1. Đọc Ground Truth
    if not os.path.exists(gt_path):
        print(f"[LỖI] Không tìm thấy file ground truth: {gt_path}")
        return []

    with open(gt_path, "r", encoding="utf-8") as f:
        gt_data = json.load(f)

    # 2. Detect tất cả vật thể một lần (reuse cho mọi threshold)
    detector = YOLODetector(model_path=model_path, conf_threshold=0.20)
    evaluator = SystemEvaluator(iou_threshold=0.5)

    all_detections = []  # [(frame_dets, roi_polygons, gt_annotations)]

    for entry in gt_data:
        image_path = entry["image_path"]
        config_path = entry.get("config_path", "configs/roi_moi.json")
        annotations = entry["annotations"]

        if not os.path.exists(image_path):
            continue

        frame = cv2.imread(image_path)
        if frame is None:
            continue

        roi_mgr = ROIManager(config_path)
        fh, fw = frame.shape[:2]
        roi_mgr.validate_resolution(fw, fh)

        dets = detector.detect(frame)
        all_detections.append((dets, roi_mgr.polygons, annotations))

    if not all_detections:
        print("[LỖI] Không có dữ liệu nào để quét ngưỡng!")
        return []

    # 3. Quét qua các ngưỡng overlap ratio
    print("=" * 70)
    print(" THỰC NGHIỆM QUÉT NGƯỠNG DIỆN TÍCH (THRESHOLD SWEEP)")
    print("=" * 70)

    sweep_results = []
    best_f1 = -1.0
    best_threshold = 0.30

    for th in thresholds:
        checker = ViolationChecker(threshold=th)
        total_boxes = 0
        total_violations = 0
        all_gt = []
        all_pred = []
        avg_overlap = []

        for dets, polys, annotations in all_detections:
            for d in dets:
                total_boxes += 1
                sp = checker.check_multi(d["box"], polys)
                avg_overlap.append(sp["overlap_pct"])
                if sp["is_violation"]:
                    total_violations += 1
                all_pred.append({
                    "box": list(d["box"]),
                    "is_violation": sp["is_violation"]
                })

            for ann in annotations:
                all_gt.append({
                    "box": ann["box"],
                    "is_violation": ann["is_violation"]
                })

        # Tính System Metrics
        if all_gt:
            metrics = evaluator.evaluate(all_gt, all_pred)
            precision = metrics["system_precision"]
            recall = metrics["system_recall"]
            f1 = metrics["system_f1"]
        else:
            precision = recall = f1 = 0.0

        mean_ov = sum(avg_overlap) / len(avg_overlap) if avg_overlap else 0.0

        if f1 > best_f1:
            best_f1 = f1
            best_threshold = th

        # Đánh giá dựa trên F1 thực tế thay vì hard-code
        sweep_results.append({
            "threshold": th,
            "threshold_pct": f"{int(th*100)}%",
            "total_boxes": total_boxes,
            "violations": total_violations,
            "legal": total_boxes - total_violations,
            "violation_rate": f"{total_violations/total_boxes*100:.1f}%" if total_boxes > 0 else "0%",
            "mean_overlap": f"{mean_ov:.1f}%",
            "precision": precision,
            "recall": recall,
            "f1": f1,
            "tp": metrics["TP"] if all_gt else 0,
            "fp": metrics["FP"] if all_gt else 0,
            "fn": metrics["FN"] if all_gt else 0
        })

        print(f"  Threshold {int(th*100):>3}% | Violations: {total_violations:>3} | "
              f"P={precision*100:5.1f}% R={recall*100:5.1f}% F1={f1*100:5.1f}%"
              f"{' ← BEST' if th == best_threshold and f1 == best_f1 else ''}")

    print(f"\n   Ngưỡng tối ưu (F1 cao nhất): {int(best_threshold*100)}% (F1 = {best_f1*100:.1f}%)")
    print("=" * 70)

    # 4. Xuất báo cáo Markdown
    os.makedirs(os.path.dirname(output_report), exist_ok=True)
    with open(output_report, "w", encoding="utf-8") as f:
        f.write("# BÁO CÁO THỰC NGHIỆM QUÉT NGƯỠNG DIỆN TÍCH (THRESHOLD SWEEP EXPERIMENT)\n\n")
        f.write("> **Định nghĩa chỉ số:** `Overlap Ratio = Diện tích giao (Bbox ∩ ROI) / Diện tích Bounding Box`.\n")
        f.write("> **Mục đích:** Chứng minh bằng số liệu thực nghiệm lý do lựa chọn ngưỡng mặc định thông qua điểm F1-Score cao nhất trên bộ Ground Truth.\n\n")

        f.write("## Bảng kết quả quét ngưỡng\n\n")
        f.write("| Ngưỡng | Số biển | Vi phạm | Hợp lệ | Tỷ lệ VP | Overlap TB | Precision | Recall | **F1-Score** | Ghi chú |\n")
        f.write("| :---: | :---: | :---: | :---: | :---: | :---: | :---: | :---: | :---: | :--- |\n")

        for r in sweep_results:
            is_best = (r["threshold"] == best_threshold)
            note = "** Tối ưu (F1 cao nhất)**" if is_best else (
                "Quá nhạy (nhiều FP)" if r["threshold"] <= 0.15 else (
                    "Quá khắt khe (bỏ sót FN)" if r["threshold"] >= 0.50 else "Chấp nhận được"
                )
            )
            f1_str = f"**{r['f1']*100:.1f}%**" if is_best else f"{r['f1']*100:.1f}%"
            f.write(f"| {r['threshold_pct']} | {r['total_boxes']} | {r['violations']} | {r['legal']} "
                    f"| {r['violation_rate']} | {r['mean_overlap']} "
                    f"| {r['precision']*100:.1f}% | {r['recall']*100:.1f}% | {f1_str} | {note} |\n")

        f.write(f"\n### Kết luận khoa học\n\n")
        f.write(f"Dựa trên thực nghiệm quét {len(thresholds)} giá trị ngưỡng từ {int(thresholds[0]*100)}% đến {int(thresholds[-1]*100)}%, "
                f"ngưỡng **{int(best_threshold*100)}%** đạt điểm F1-Score cao nhất **({best_f1*100:.1f}%)**, "
                f"cân bằng tối ưu giữa:\n\n")
        f.write(f"1. **Precision** (tỷ lệ cảnh báo đúng): tránh báo động giả cho các biển chỉ nhô nhẹ mép.\n")
        f.write(f"2. **Recall** (tỷ lệ phát hiện): không bỏ sót các trường hợp lấn chiếm thực sự.\n\n")
        f.write(f"Do đó, ngưỡng `{int(best_threshold*100)}%` được chọn làm giá trị mặc định của hệ thống.\n")

    print(f"\n[+] Đã hoàn thành thực nghiệm Threshold Sweep! Báo cáo: {output_report}")
    return sweep_results


if __name__ == "__main__":
    run_threshold_sweep()

FILE: tracking/tracker.py
MỤC ĐÍCH KIỂM TRA: Wrapper YOLO + ByteTrack, tiền xử lý, hậu xử lý, xử lý track_id = -1

"""
==============================================================================
ĐỒ ÁN TỐT NGHIỆP: PHÁT HIỆN BIỂN HIỆU/BIỂN QUẢNG CÁO LẤN CHIẾM VỈA HÈ QUA CCTV
Chủ đề: Object Tracking & Multi-Object Association
TASK: 21 - Xây dựng Module tracking/tracker.py tích hợp ByteTrack
==============================================================================
"""

import os
import cv2
import numpy as np
from ultralytics import YOLO


class ObjectTracker:
    """
    Bộ bám vết vật thể (Multi-Object Tracker) tích hợp thuật toán ByteTrack.
    Nhiệm vụ: Gán và duy trì một mã định danh duy nhất (track_id: 1, 2, 3...)
    cho từng biển hiệu xuyên suốt các khung hình video theo thời gian.
    """
    def __init__(self, model_path: str = "models/best.pt", conf_threshold: float = 0.25, tracker_type: str = "bytetrack.yaml", imgsz: int = 640):
        """
        Khởi tạo ObjectTracker.
        
        Args:
            model_path: Đường dẫn file trọng số YOLO (vd: 'models/best.pt', 'models/best.onnx')
            conf_threshold: Ngưỡng độ tin cậy tối thiểu (mặc định 0.25 = 25% theo chuẩn đánh giá hệ thống)
            tracker_type: File cấu hình tracker ('bytetrack.yaml' hoặc 'botsort.yaml')
            imgsz: Kích thước cạnh ảnh chuẩn khi suy luận (mặc định 640)
        """
        self.model_path = model_path
        self.conf_threshold = conf_threshold
        self.tracker_type = tracker_type
        self.imgsz = imgsz
        
        if not os.path.exists(model_path) and not model_path.endswith("yolov8n.pt"):
            raise FileNotFoundError(f"Không tìm thấy file trọng số mô hình: {model_path}")
            
        print(f"[OBJECT TRACKER] Nạp mô hình: {model_path} | Thuật toán: {tracker_type} (Conf: {conf_threshold}, Imgsz: {imgsz})")
        self.model = YOLO(model_path)
        self.class_names = self.model.names

    def track(self, frame, conf: float = None, imgsz: int = None) -> list:
        """
        Thực hiện phát hiện và bám vết đối tượng trong khung hình.
        
        Args:
            frame: Ảnh ma trận BGR nguyên bản từ camera (Raw Frame)
            conf: Tùy chọn override ngưỡng tin cậy (mặc định dùng self.conf_threshold)
            imgsz: Tùy chọn override kích thước suy luận (mặc định dùng self.imgsz)
            
        Returns:
            list: Danh sách các đối tượng được bám vết:
                  [
                      {
                          "track_id": int (Mã định danh duy nhất: 1, 2, 3...; -1 nếu chưa khớp),
                          "box": (x1, y1, x2, y2),
                          "conf": float (0.0 -> 1.0),
                          "class_id": int,
                          "class_name": str
                      },
                      ...
                  ]
        """
        target_conf = conf if conf is not None else self.conf_threshold
        target_imgsz = imgsz if imgsz is not None else self.imgsz
        # persist=True: Bắt buộc để ByteTrack nhớ trạng thái Kalman Filter giữa các frames liên tiếp
        results = self.model.track(
            source=frame,
            persist=True,
            tracker=self.tracker_type,
            conf=target_conf,
            imgsz=target_imgsz,
            verbose=False
        )

        tracked_objects = []
        if len(results) == 0 or results[0].boxes is None:
            return tracked_objects

        boxes = results[0].boxes
        for box in boxes:
            xyxy = box.xyxy[0].cpu().numpy().astype(int)
            conf = float(box.conf[0].cpu().numpy())
            cls_id = int(box.cls[0].cpu().numpy())
            cls_name = self.class_names.get(cls_id, f"class_{cls_id}")

            # Lấy track_id do ByteTrack sinh ra
            if box.id is not None:
                track_id = int(box.id[0].cpu().numpy())
            else:
                track_id = -1  # Vật thể mới xuất hiện chưa đủ độ tin cậy để gán ID

            tracked_objects.append({
                "track_id": track_id,
                "box": (int(xyxy[0]), int(xyxy[1]), int(xyxy[2]), int(xyxy[3])),
                "conf": round(conf, 2),
                "class_id": cls_id,
                "class_name": cls_name
            })

        return tracked_objects

    def reset(self):
        """Reset bộ nhớ trạng thái tracker (khi chuyển luồng camera hoặc tua lại video)"""
        # Khởi tạo lại tracker state trong model
        if hasattr(self.model, 'predictor') and self.model.predictor is not None:
            pred = self.model.predictor
            if hasattr(pred, 'trackers') and pred.trackers is not None:
                for trk in pred.trackers:
                    if hasattr(trk, 'reset'):
                        try:
                            trk.reset()
                        except Exception:
                            pass
            else:
                if hasattr(pred, 'trackers'):
                    try:
                        del pred.trackers
                    except Exception:
                        pass


================================================================================
FILE: tracking/verifier.py
MỤC ĐÍCH KIỂM TRA: Bộ xác minh chuỗi thời gian, Perspective-Adaptive Stationary Check, Cooldown
================================================================================

"""
==============================================================================
ĐỒ ÁN TỐT NGHIỆP: PHÁT HIỆN BIỂN HIỆU/BIỂN QUẢNG CÁO LẤN CHIẾM VỈA HÈ QUA CCTV
Chủ đề: Temporal Reasoning & False Positive Elimination
TASK: 22 - Xây dựng Module tracking/verifier.py (Bộ lọc xác nhận theo thời gian)

v2.2 – Nâng cấp hoàn thiện bảo vệ ĐATN:
  - Chuyển từ displacement-from-origin sang cửa sổ trượt (sliding window 1.0s)
    để không từ chối biển được đặt xuống rồi đứng yên.
  - Bổ sung cơ chế Perspective-Adaptive Stationary Check (chuẩn hóa ngưỡng dịch
    chuyển theo kích thước đường chéo hộp giới hạn kết hợp trần max_displacement_pixels).
  - Tính thời gian vi phạm bằng giây thực (time.time() / simulated_ts) thay vì
    số frame, bất biến với frame_skip và biến thiên FPS.
==============================================================================
"""

import time
from collections import deque


class TrackState:
    """Các trạng thái vòng đời của một vật thể biển hiệu được theo dõi."""
    NORMAL = "NORMAL"          # Biển hợp lệ, không lấn chiếm vỉa hè
    SUSPECTED = "SUSPECTED"    # Nghi vấn vi phạm (< N giây liên tiếp, chưa phạt)
    CONFIRMED = "CONFIRMED"    # ĐÃ XÁC NHẬN VI PHẠM (>= N giây liên tiếp)


class TemporalVerifier:
    """
    Bộ lọc xác nhận vi phạm theo chuỗi thời gian (Temporal Verification).

    Nguyên lý: Chỉ kết luận và phát cảnh báo vi phạm khi một đối tượng (track_id)
    duy trì hành vi lấn chiếm vỉa hè liên tục trong tối thiểu N GIÂY THỰC.

    Cơ chế sàng lọc:
      - Người đi bộ bê biển đi ngang qua vỉa hè (sliding window displacement).
      - Xe cộ vô tình che khuất chớp nhoáng (Flickering) với tolerance_frames.
      - Cảnh báo trùng lặp (Spam Alert) nhờ cơ chế Cooldown Timer & Spatial Radius.
    """

    def __init__(
        self,
        confirm_frames: int = 20,
        confirm_seconds: float = 3.0,
        cooldown_seconds: float = 60.0,
        max_missing_frames: int = 30,
        tolerance_frames: int = 2,
        source_fps: float = None,
        min_avg_confidence: float = 0.40,
        require_stationary: bool = True,
        max_displacement_pixels: float = 50.0,
        stationary_window_seconds: float = 1.0,
        spatial_cooldown_radius: float = 50.0,
        adaptive_perspective_scale: bool = True,
    ):
        """
        Args:
            confirm_frames:            Số frames liên tiếp vi phạm tối thiểu (fallback).
            confirm_seconds:           Thời gian vi phạm liên tiếp tối thiểu (giây).
            cooldown_seconds:          Thời gian chờ (giây) chống gửi cảnh báo trùng lặp.
            max_missing_frames:        Xóa track khỏi RAM nếu mất dấu quá số frame này.
            tolerance_frames:          Dung sai frame rung lắc/flicker trước khi reset bộ đếm.
            source_fps:                FPS của nguồn video (dùng để quy đổi giây <-> frame).
            min_avg_confidence:        Ngưỡng confidence TB tối thiểu để CONFIRMED.
            require_stationary:        Bật/tắt bộ lọc yêu cầu vật thể đứng yên.
            max_displacement_pixels:   Độ dời tối đa (trần) trong cửa sổ để coi là đứng yên.
            stationary_window_seconds: Độ rộng cửa sổ trượt kiểm tra đứng yên (mặc định 1.0s).
            spatial_cooldown_radius:   Bán kính không gian (pixels) khử trùng lặp khi nhảy ID.
            adaptive_perspective_scale: Tự động co ngưỡng dịch chuyển theo kích thước BBox ở xa.
        """
        self.confirm_frames = confirm_frames
        self.confirm_seconds = confirm_seconds
        self.cooldown_seconds = cooldown_seconds
        self.max_missing_frames = max_missing_frames
        self.tolerance_frames = tolerance_frames
        self.source_fps = source_fps

        self.min_avg_confidence = min_avg_confidence

        # --- Stationary check (sliding window + perspective-adaptive) ---
        self.require_stationary = require_stationary
        self.max_displacement_pixels = max_displacement_pixels
        self.stationary_window_seconds = stationary_window_seconds
        self.adaptive_perspective_scale = adaptive_perspective_scale

        # Spatial deduplication khi track_id bị đổi
        self.spatial_cooldown_radius = spatial_cooldown_radius
        self.confirmed_locations: list = []     # [{cx, cy, time, track_id}, ...]

        self.track_records: dict = {}
        self.current_frame_index = 0

    # ------------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------------

    def _elapsed_viol_seconds(self, record: dict, now: float) -> float:
        """Trả về số giây vi phạm liên tiếp kể từ lần đầu bắt đầu của chuỗi quan sát hiện tại."""
        if not record.get("viol_confs"):
            return 0.0
        start = record.get("viol_start_time")
        if start is None:
            return 0.0
        return max(0.0, now - start)

    def _is_stationary(self, record: dict, now: float) -> bool:
        """
        Kiểm tra cửa sổ trượt: lấy tất cả vị trí trong `stationary_window_seconds`
        gần nhất, tính khoảng cách cực đại giữa các vị trí trong cửa sổ.
        Nếu bật `adaptive_perspective_scale`, ngưỡng dịch chuyển tối đa được chuẩn hóa
        theo đường chéo trung bình của Bounding Box (50% đường chéo, tối thiểu 15px,
        tối đa `max_displacement_pixels`), giúp chống sai lệch phối cảnh ở cự ly xa.
        """
        if not self.require_stationary:
            return True

        window = record.get("pos_window")
        if window is None or len(window) < 2:
            return True  # Không đủ dữ liệu -> giả định đứng yên

        cutoff = now - self.stationary_window_seconds
        recent = [item for item in window if item[0] >= cutoff]
        if len(recent) < 2:
            return True

        xs = [p[1] for p in recent]
        ys = [p[2] for p in recent]
        max_d = max(
            ((xs[i] - xs[j]) ** 2 + (ys[i] - ys[j]) ** 2) ** 0.5
            for i in range(len(recent))
            for j in range(i + 1, len(recent))
        )

        effective_threshold = self.max_displacement_pixels
        if self.adaptive_perspective_scale and len(recent[-1]) >= 4:
            avg_diag = sum(p[3] for p in recent) / len(recent)
            if avg_diag > 0:
                effective_threshold = min(
                    self.max_displacement_pixels,
                    max(15.0, avg_diag * 0.50)
                )

        return max_d <= effective_threshold

    def _confirm_threshold_reached(self, record: dict, now: float) -> bool:
        """
        Kiểm tra đã đạt ngưỡng xác nhận chưa.
        Ưu tiên: giây thực nếu có thể tính được (viol_start_time tồn tại),
        fallback về số frame.
        """
        if record.get("viol_start_time") is not None and record.get("viol_confs"):
            return self._elapsed_viol_seconds(record, now) >= self.confirm_seconds
        return record["count"] >= self.confirm_frames

    # ------------------------------------------------------------------
    # public API
    # ------------------------------------------------------------------

    def process_frame(
        self,
        tracked_objects: list,
        checker,
        polygons: list,
        current_timestamp: float = None,
        precomputed_spatial: list = None,
    ) -> list:
        """
        Phân tích chuỗi thời gian cho danh sách đối tượng trong khung hình hiện tại.

        Args:
            tracked_objects:     Danh sách đối tượng từ ObjectTracker (box, conf, track_id)
            checker:             ViolationChecker để kiểm tra hình học không gian
            polygons:            Danh sách đa giác vỉa hè (Multi-ROI)
            current_timestamp:   Mốc thời gian thực (time.time()); mặc định lấy từ hệ thống
            precomputed_spatial: Kết quả spatial đã tính trước (tránh tính 2 lần trong benchmark)
        """
        if current_timestamp is None:
            current_timestamp = time.time()

        self.current_frame_index += 1
        active_ids: set = set()
        enriched_results: list = []

        for idx, obj in enumerate(tracked_objects):
            track_id = obj.get("track_id", -1)
            box = obj["box"]
            conf = obj["conf"]

            # 1. Kiểm tra hình học không gian
            if precomputed_spatial is not None and idx < len(precomputed_spatial):
                spatial_res = precomputed_spatial[idx]
            else:
                spatial_res = checker.check_multi(box, polygons)
            is_spatial_viol = spatial_res["is_violation"]

            # Tính tọa độ tâm và đường chéo hộp giới hạn
            cx = (box[0] + box[2]) / 2.0
            cy = (box[1] + box[3]) / 2.0
            box_diag = ((box[2] - box[0]) ** 2 + (box[3] - box[1]) ** 2) ** 0.5

            # Vật thể chưa có track_id hợp lệ (frame đầu tiên xuất hiện)
            if track_id <= 0:
                enriched_results.append({
                    "track_id": track_id,
                    "box": box,
                    "conf": conf,
                    "spatial_res": spatial_res,
                    "temporal_status": TrackState.SUSPECTED if is_spatial_viol else TrackState.NORMAL,
                    "violation_frames": 1 if is_spatial_viol else 0,
                    "is_new_alert": False,
                })
                continue

            active_ids.add(track_id)

            # 2. Khởi tạo bản ghi nếu là ID mới
            if track_id not in self.track_records:
                self.track_records[track_id] = {
                    "count": 0,
                    "viol_start_time": None,
                    "last_viol_timestamp": None,
                    "status": TrackState.NORMAL,
                    "last_seen": self.current_frame_index,
                    "last_alert_time": 0.0,
                    "non_viol_streak": 0,
                    "viol_confs": deque(maxlen=200),   # deque lưu (timestamp, conf)
                    "pos_window": deque(maxlen=200),   # deque lưu (timestamp, cx, cy, box_diag)
                }

            record = self.track_records[track_id]
            record["last_seen"] = self.current_frame_index

            # Cập nhật sliding window vị trí kèm đường chéo BBox
            record["pos_window"].append((current_timestamp, cx, cy, box_diag))

            # 3. Cập nhật bộ đếm thời gian
            is_new_alert = False
            suspect_reason = ""

            if is_spatial_viol:
                # Kiểm tra gián đoạn chuỗi vi phạm (gap lớn hơn dung sai liên tục)
                frame_interval = (1.0 / self.source_fps) if self.source_fps else 0.04
                max_allowed_gap = max(0.6, (self.tolerance_frames + 2) * frame_interval * 3)
                if record["last_viol_timestamp"] is not None:
                    gap = current_timestamp - record["last_viol_timestamp"]
                    if gap > max_allowed_gap:
                        # Gián đoạn quá lâu -> reset chuỗi quan sát liên tục
                        record["viol_confs"].clear()
                        record["viol_start_time"] = current_timestamp
                        record["count"] = 0

                if record["viol_start_time"] is None:
                    record["viol_start_time"] = current_timestamp

                record["last_viol_timestamp"] = current_timestamp
                record["viol_confs"].append((current_timestamp, conf))
                record["count"] = len(record["viol_confs"])
                record["non_viol_streak"] = 0

                # Tính confidence trung bình trên chuỗi quan sát thực tế (không bị thổi phồng khi decay)
                confs = [c for t, c in record["viol_confs"]]
                avg_conf = (sum(confs) / len(confs)) if confs else 0.0

                # Kiểm tra đã đạt ngưỡng thời gian/frame chưa
                if self._confirm_threshold_reached(record, current_timestamp):
                    # Stationary check (sliding window)
                    is_stationary = self._is_stationary(record, current_timestamp)

                    if not is_stationary:
                        # Đang di chuyển -> người bê biển đi ngang, giữ SUSPECTED
                        record["status"] = TrackState.SUSPECTED
                        suspect_reason = "DI CHUYEN"
                    elif avg_conf < self.min_avg_confidence:
                        # Confidence TB quá thấp -> giữ SUSPECTED
                        record["status"] = TrackState.SUSPECTED
                        suspect_reason = f"CONF {int(avg_conf * 100)}%<{int(self.min_avg_confidence * 100)}%"
                    else:
                        record["status"] = TrackState.CONFIRMED
                        suspect_reason = ""

                    # Spatial & temporal deduplication
                    if record["status"] == TrackState.CONFIRMED:
                        is_spatially_cooldown = any(
                            ((cx - c["cx"]) ** 2 + (cy - c["cy"]) ** 2) ** 0.5 <= self.spatial_cooldown_radius
                            and (current_timestamp - c["time"]) < self.cooldown_seconds
                            for c in self.confirmed_locations
                        )
                        time_ok = (current_timestamp - record["last_alert_time"]) >= self.cooldown_seconds

                        if not is_spatially_cooldown and time_ok:
                            is_new_alert = True
                            record["last_alert_time"] = current_timestamp
                            self.confirmed_locations.append({
                                "cx": cx,
                                "cy": cy,
                                "time": current_timestamp,
                                "track_id": track_id,
                            })
                            # Dọn dẹp mốc cũ
                            self.confirmed_locations = [
                                c for c in self.confirmed_locations
                                if (current_timestamp - c["time"]) <= self.cooldown_seconds * 2
                            ]
                else:
                    if record["status"] != TrackState.CONFIRMED:
                        record["status"] = TrackState.SUSPECTED
                    suspect_reason = ""

            else:
                # Không vi phạm ở frame này
                record["non_viol_streak"] += 1
                if record["non_viol_streak"] > self.tolerance_frames:
                    # Vượt ngưỡng dung sai -> reset hoàn toàn
                    record["count"] = 0
                    record["viol_confs"].clear()
                    record["viol_start_time"] = None
                    record["last_viol_timestamp"] = None
                    record["status"] = TrackState.NORMAL
                else:
                    # Trong ngưỡng dung sai (1-2 frame flicker): trừ conf tương ứng khi decay
                    if record["viol_confs"]:
                        record["viol_confs"].popleft()
                    record["count"] = len(record["viol_confs"])
                    if record["count"] == 0:
                        record["viol_start_time"] = None
                        record["last_viol_timestamp"] = None
                        record["status"] = TrackState.NORMAL

            enriched_results.append({
                "track_id": track_id,
                "box": box,
                "conf": conf,
                "spatial_res": spatial_res,
                "temporal_status": record["status"],
                "suspect_reason": suspect_reason,
                "violation_frames": record["count"],
                "violation_seconds": round(self._elapsed_viol_seconds(record, current_timestamp), 2),
                "is_new_alert": is_new_alert,
            })

        # 4. Decay các track bị che khuất (Missing Occlusion Decay)
        for tid, r in self.track_records.items():
            if tid not in active_ids:
                missing_streak = self.current_frame_index - r["last_seen"]
                if missing_streak > self.tolerance_frames:
                    if r["viol_confs"]:
                        r["viol_confs"].popleft()  # Trừ conf tương ứng khi decay
                    r["count"] = len(r["viol_confs"])
                    if r["count"] == 0:
                        r["viol_start_time"] = None
                        r["last_viol_timestamp"] = None
                        r["status"] = TrackState.NORMAL

        # 5. Garbage Collection – xóa track cũ khỏi RAM
        stale_ids = [
            tid for tid, r in self.track_records.items()
            if (self.current_frame_index - r["last_seen"]) > self.max_missing_frames
        ]
        for tid in stale_ids:
            del self.track_records[tid]

        return enriched_results

    def reset(self):
        """Xóa toàn bộ bản ghi trạng thái (khi tua video hoặc đổi camera)."""
        self.track_records.clear()
        self.confirmed_locations.clear()
        self.current_frame_index = 0
