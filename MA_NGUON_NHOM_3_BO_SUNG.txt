    <script>
        let currentStatusFilter = 'all';
        let currentDateFilter = 'all'; // Mặc định hiển thị tất cả hồ sơ để vi phạm mới phát sinh luôn hiển thị ngay
        let allAvailableDates = [];
        let activeViolIdForDismiss = null;
        let activeViolIdForConfirm = null;

        function formatDateVN(dateStr) {
            if (!dateStr || dateStr === 'all') return 'Tất cả các ngày';
            const parts = dateStr.split('-');
            if (parts.length === 3) return `${parts[2]}/${parts[1]}/${parts[0]}`;
            return dateStr;
        }

        function isTodayDate(dateStr) {
            const today = new Date();
            const y = today.getFullYear();
            const m = String(today.getMonth() + 1).padStart(2, '0');
            const d = String(today.getDate()).padStart(2, '0');
            return dateStr === `${y}-${m}-${d}`;
        }

        async function loadDatesList() {
            try {
                const res = await fetch('/api/dates');
                allAvailableDates = await res.json();

                // Nếu ngày mặc định chưa có dữ liệu và danh sách có ngày khác, chọn ngày mới nhất
                if (allAvailableDates && allAvailableDates.length > 0) {
                    const hasSelected = allAvailableDates.some(item => item.date === currentDateFilter);
                    if (!hasSelected && currentDateFilter !== 'all') {
                        currentDateFilter = allAvailableDates[0].date;
                    }
                }

                renderDatePills();
            } catch (e) {
                console.error('Lỗi nạp danh sách ngày:', e);
            }
        }

        function renderDatePills() {
            const container = document.getElementById('date-pills-container');
            if (!container) return;

            if (!allAvailableDates || allAvailableDates.length === 0) {
                container.innerHTML = `<span style="font-size: 0.78rem; color: var(--text-dim);">Chưa có dữ liệu ngày</span>`;
                return;
            }

            const pillsHtml = allAvailableDates.slice(0, 5).map(item => {
                const isAct = (currentDateFilter === item.date) ? 'active' : '';
                const isToday = isTodayDate(item.date) || item.date === '2026-10-19';
                const label = isToday ? `Hôm nay (${formatDateVN(item.date)})` : formatDateVN(item.date);
                return `
                    <button class="date-pill-btn ${isAct}" onclick="filterByDate('${item.date}')" title="Chỉ xem hồ sơ ngày ${formatDateVN(item.date)}">
                        <i class="fa-solid fa-calendar-day"></i> ${label}
                        <span class="date-pill-count">${item.count}</span>
                    </button>
                `;
            }).join('');

            container.innerHTML = pillsHtml;

            // Nút Xem Tất Cả
            const btnAll = document.getElementById('btn-all-dates');
            if (btnAll) {
                btnAll.classList.toggle('active', currentDateFilter === 'all');
            }

            // Input date picker
            const dateInput = document.getElementById('date-picker-input');
            if (dateInput && currentDateFilter !== 'all') {
                dateInput.value = currentDateFilter;
            }
        }

        function filterByDate(dateStr) {
            currentDateFilter = dateStr;
            renderDatePills();
            loadData();
        }

        function onDatePickerChange(val) {
            if (val) {
                filterByDate(val);
            }
        }

        async function loadData() {
            try {
                // Nạp thống kê KPI theo ngày lọc (nếu chọn)
                const statsUrl = currentDateFilter === 'all' ? '/api/stats' : `/api/stats?date=${encodeURIComponent(currentDateFilter)}`;
                const statsRes = await fetch(statsUrl);
                const stats = await statsRes.json();

                // Update KPIs
                const byStatus = stats.violations_by_status || {};
                const total = stats.total_violations || 0;
                const pending = byStatus.pending || 0;
                const confirmed = byStatus.confirmed || 0;
                const dismissed = byStatus.dismissed || 0;
                const resolved = byStatus.resolved || 0;

                document.getElementById('total-val').innerText = total;
                document.getElementById('pending-val').innerText = pending;
                document.getElementById('confirmed-val').innerText = confirmed;
                document.getElementById('dismissed-val').innerText = dismissed;
                document.getElementById('resolved-val').innerText = resolved;

                document.getElementById('count-all').innerText = total;
                document.getElementById('count-pending').innerText = pending;
                document.getElementById('count-confirmed').innerText = confirmed;
                document.getElementById('count-dismissed').innerText = dismissed;
                document.getElementById('count-resolved').innerText = resolved;

                // Load violations with filter: status & date
                let url = '/api/violations?';
                if (currentStatusFilter !== 'all') url += `status=${encodeURIComponent(currentStatusFilter)}&`;
                if (currentDateFilter !== 'all') url += `date=${encodeURIComponent(currentDateFilter)}&`;

                const violsRes = await fetch(url);
                const viols = await violsRes.json();

                const container = document.getElementById('violations-container');
                if (!viols || viols.length === 0) {
                    const dateMsg = currentDateFilter === 'all' ? '' : ` trong ngày ${formatDateVN(currentDateFilter)}`;
                    container.innerHTML = `
                        <div class="empty-state" style="grid-column: 1/-1;">

        /* API Call */
        async function updateStatusDirect(id, status, reviewer, reason) {
            try {
                const res = await fetch(`/api/violations/${id}/status`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        status: status,
                        reviewed_by: reviewer,
                        dismiss_reason: reason
                    })
                });
                if (res.ok) {
                    loadData();
                } else {
                    alert('Lỗi cập nhật trạng thái vi phạm!');
                }
            } catch (err) {
                console.error('Lỗi API:', err);
                alert('Không thể kết nối đến máy chủ.');
            }
        }

        async function selectVideo(fname) {
            const vid = allVideosList.find(v => v.filename === fname);
            if (!vid) return;

            currentVideo = vid;

            // Highlight card
            document.querySelectorAll('.roi-video-card').forEach(c => c.classList.remove('active'));
            const idx = allVideosList.findIndex(v => v.filename === fname);
            const activeEl = document.getElementById(`vid-card-${idx}`);
            if (activeEl) activeEl.classList.add('active');

            // Show loader
            const loader = document.getElementById('roi-canvas-loader');
            if (loader) loader.style.display = 'flex';

            activePolygon = [];
            completedPolygons = [];

            try {
                const [configRes, imgRes] = await Promise.all([
                    fetch(`/api/roi/config?video=${encodeURIComponent(fname)}`).then(r => r.json()),
                    fetch(`/api/roi/frame?video=${encodeURIComponent(fname)}`)
                ]);

                if (!imgRes.ok) throw new Error("Không thể trích xuất khung hình từ video");

                const blob = await imgRes.blob();
                const imgUrl = URL.createObjectURL(blob);
                const img = new Image();

                img.onload = () => {
                    currentImage = img;
                    canvas.width = img.naturalWidth || vid.width || 1280;
                    canvas.height = img.naturalHeight || vid.height || 720;

                    if (configRes && configRes.polygons && configRes.polygons.length > 0) {
                        completedPolygons = JSON.parse(JSON.stringify(configRes.polygons));
                    }

                    // Cập nhật toolbar
                    document.getElementById('roi-current-video-title').innerHTML = `<i class="fa-solid fa-video"></i> ${vid.filename}`;
                    document.getElementById('roi-current-meta').innerText = `${canvas.width} × ${canvas.height} px (${vid.fps} FPS)`;

                    document.getElementById('roi-canvas-placeholder').style.display = 'none';
                    canvas.style.display = 'block';
                    if (loader) loader.style.display = 'none';

                    updateRegionsSummary();
                    redrawCanvas();
                };
                img.src = imgUrl;
            } catch (err) {
                console.error('Lỗi selectVideo:', err);
                if (loader) loader.style.display = 'none';
                alert('Không thể mở video: ' + err.message);
            }
        }

        function getCanvasCoords(e) {
            const rect = canvas.getBoundingClientRect();
            const scaleX = canvas.width / rect.width;
            const scaleY = canvas.height / rect.height;
            let x = Math.round((e.clientX - rect.left) * scaleX);
            let y = Math.round((e.clientY - rect.top) * scaleY);
            x = Math.max(0, Math.min(canvas.width, x));
            y = Math.max(0, Math.min(canvas.height, y));
            return { x, y };
        }

        function redrawCanvas() {
            if (!ctx || !currentImage) return;

            ctx.clearRect(0, 0, canvas.width, canvas.height);
            ctx.drawImage(currentImage, 0, 0, canvas.width, canvas.height);

            // 1. Vẽ các polygon đã chốt
            completedPolygons.forEach((poly, polyIdx) => {
                if (!poly || poly.length < 3) return;
                const color = REGION_PALETTE[polyIdx % REGION_PALETTE.length];

                ctx.beginPath();
                ctx.moveTo(poly[0][0], poly[0][1]);
                for (let i = 1; i < poly.length; i++) {
                    ctx.lineTo(poly[i][0], poly[i][1]);
                }
                ctx.closePath();
                ctx.fillStyle = color.fill;
                ctx.fill();
                ctx.strokeStyle = color.stroke;
                ctx.lineWidth = 3;
                ctx.stroke();

                // Vẽ các đỉnh
                poly.forEach((pt, ptIdx) => {
                    ctx.beginPath();
                    ctx.arc(pt[0], pt[1], 5, 0, Math.PI * 2);
                    ctx.fillStyle = '#ffffff';
                    ctx.fill();
                    ctx.strokeStyle = color.stroke;
                    ctx.lineWidth = 2;
                    ctx.stroke();
                });

                // Vẽ nhãn vùng
                const firstPt = poly[0];
                const labelText = `[VỈA HÈ #${polyIdx + 1}] (${poly.length} đỉnh)`;
                ctx.font = 'bold 15px Inter, sans-serif';
                const tw = ctx.measureText(labelText).width;
                const lx = firstPt[0] + 8;
                const ly = Math.max(22, firstPt[1] - 12);

                ctx.fillStyle = 'rgba(15, 23, 42, 0.88)';
                ctx.fillRect(lx, ly - 18, tw + 14, 24);
                ctx.strokeStyle = color.stroke;
                ctx.lineWidth = 1;
                ctx.strokeRect(lx, ly - 18, tw + 14, 24);
                ctx.fillStyle = color.text;
                ctx.fillText(labelText, lx + 7, ly);
            });

            // 2. Vẽ polygon đang chấm dở
            if (activePolygon.length > 0) {
                ctx.beginPath();
                ctx.moveTo(activePolygon[0][0], activePolygon[0][1]);
                for (let i = 1; i < activePolygon.length; i++) {
                    ctx.lineTo(activePolygon[i][0], activePolygon[i][1]);
                }
                if (mousePos) {
                    ctx.lineTo(mousePos.x, mousePos.y);
                }
                ctx.strokeStyle = '#f59e0b';
                ctx.lineWidth = 2.5;
                ctx.setLineDash([6, 4]);
                ctx.stroke();
                ctx.setLineDash([]);

                if (activePolygon.length >= 3) {
                    ctx.closePath();
                    ctx.fillStyle = 'rgba(245, 158, 11, 0.18)';
                    ctx.fill();
                }

                activePolygon.forEach((pt, ptIdx) => {
                    ctx.beginPath();
                    ctx.arc(pt[0], pt[1], 6, 0, Math.PI * 2);
                    ctx.fillStyle = '#ef4444';
                    ctx.fill();
                    ctx.strokeStyle = '#ffffff';
                    ctx.lineWidth = 2;
                    ctx.stroke();

                    ctx.font = 'bold 13px Inter, sans-serif';
                    ctx.fillStyle = '#f59e0b';
                    ctx.fillText(`P${ptIdx + 1}`, pt[0] + 8, pt[1] - 5);
                });
            }
        }

        function finishCurrentPolygon() {
            if (activePolygon.length < 3) {
                showToast('Một vùng vỉa hè cần tối thiểu 3 đỉnh!', true);
                return;
            }
            completedPolygons.push([...activePolygon]);
            activePolygon = [];
            updateRegionsSummary();
            redrawCanvas();
            showToast(`Đã chốt Vỉa hè #${completedPolygons.length}! Bạn có thể vẽ tiếp vùng khác.`);
        }

        function undoVertex() {
            if (activePolygon.length > 0) {
                activePolygon.pop();
                redrawCanvas();
            } else if (completedPolygons.length > 0) {
                activePolygon = completedPolygons.pop();
                updateRegionsSummary();
                redrawCanvas();
                showToast('Đã mở lại vùng trước để chỉnh sửa tiếp.');
            }
        }

        function clearCurrentPolygon() {
            if (activePolygon.length > 0) {
                activePolygon = [];
                redrawCanvas();
                showToast('Đã xóa vùng đang vẽ dở.');
            }
        }

        function clearAllPolygons() {
            if (completedPolygons.length === 0 && activePolygon.length === 0) return;
            if (confirm('Bạn có chắc chắn muốn xóa TẤT CẢ các vùng vỉa hè của video này?')) {
                completedPolygons = [];
                activePolygon = [];
                updateRegionsSummary();
                redrawCanvas();
                showToast('Đã xóa tất cả các vùng.');
            }
        }

        function deleteRegion(idx) {
            if (idx >= 0 && idx < completedPolygons.length) {
                completedPolygons.splice(idx, 1);
                updateRegionsSummary();
                redrawCanvas();
                showToast(`Đã xóa vùng vỉa hè #${idx + 1}.`);
            }
        }

        function updateRegionsSummary() {
            const container = document.getElementById('roi-regions-summary');
            if (!container) return;

            if (completedPolygons.length === 0) {
                container.innerHTML = `<span style="font-size: 0.78rem; color: var(--text-dim);">Chưa chốt vùng nào</span>`;
                return;
            }

            container.innerHTML = completedPolygons.map((poly, idx) => {
                const color = REGION_PALETTE[idx % REGION_PALETTE.length];
                return `
                    <div class="region-badge-pill">
                        <span class="color-dot" style="background: ${color.dot};"></span>
                        <span>Vỉa hè #${idx + 1} (${poly.length} đỉnh)</span>
                        <button class="region-delete-btn" onclick="deleteRegion(${idx})" title="Xóa vùng này">
                            <i class="fa-solid fa-xmark"></i>
                        </button>
                    </div>
                `;
            }).join('');
        }

        async function saveRoiConfigToServer() {
            if (!currentVideo) {
                alert('Vui lòng chọn 1 video trước khi lưu!');
                return;
            }

            if (activePolygon.length >= 3) {
                completedPolygons.push([...activePolygon]);
                activePolygon = [];
            }

            if (completedPolygons.length === 0) {
                alert('Chưa có vùng vỉa hè nào được vẽ! Hãy chấm ít nhất 3 điểm trên khung hình.');
                return;
            }

            try {
                const payload = {
                    video: currentVideo.filename,
                    polygons: completedPolygons,
                    resolution: {
                        width: canvas.width,
                        height: canvas.height
                    }
                };

                const res = await fetch('/api/roi/save', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const data = await res.json();
                if (res.ok && data.ok) {
                    showToast(`Đã lưu thành công ${data.total_regions} vùng vào ${data.config_filename}!`);

                    // Cập nhật trạng thái trong video list
                    currentVideo.has_roi = true;
                    currentVideo.regions_count = data.total_regions;
                    currentVideo.points_count = data.total_points;

                    renderRoiVideoList(allVideosList);
                    const configured = allVideosList.filter(v => v.has_roi).length;
                    document.getElementById('roi-configured-count-badge').innerText = configured;

                    updateRegionsSummary();
                    redrawCanvas();
                } else {
                    alert('Lỗi lưu cấu hình: ' + (data.error || 'Không xác định'));
                }
            } catch (err) {
                console.error('Lỗi lưu cấu hình:', err);
                alert('Lỗi kết nối máy chủ khi lưu cấu hình.');
            }
        }

        function showToast(msg, isWarn = false) {
            const toast = document.getElementById('roi-toast');
            if (!toast) return;
            toast.style.background = isWarn ? '#f59e0b' : '#10b981';
            toast.innerHTML = `<i class="fa-solid ${isWarn ? 'fa-triangle-exclamation' : 'fa-circle-check'}"></i> ${msg}`;
            toast.style.display = 'flex';
            clearTimeout(toast._timer);
            toast._timer = setTimeout(() => {
                toast.style.display = 'none';
            }, 3500);
        }

                if (e.button === 0) { // Chuột trái: Thêm đỉnh
                    activePolygon.push([coords.x, coords.y]);
                    redrawCanvas();
                } else if (e.button === 2) { // Chuột phải: Undo đỉnh

                    undoVertex();

                if (!currentImage) return;

                document.getElementById('roi-cursor-coord').innerText = `X: ${coords.x} | Y: ${coords.y}`;
                if (activePolygon.length > 0) {
                    redrawCanvas();
                }
            });

            canvas.addEventListener('mouseleave', () => {
                mousePos = null;
                if (activePolygon.length > 0) redrawCanvas();
            });

                e.preventDefault();

        function selectAndPlayVideo(videoName) {
            currentStreamVideo = videoName;
            currentStreamToken = Date.now();

            // Cập nhật giao diện thẻ được chọn
            document.querySelectorAll('.player-video-card').forEach(c => {
                if (c.getAttribute('data-video') === videoName) {
                    c.classList.add('active');
                } else {
                    c.classList.remove('active');
                }
            });

            // Ẩn placeholder
            const placeholder = document.getElementById('player-screen-placeholder');
            if (placeholder) placeholder.style.display = 'none';

            // Cập nhật tiêu đề HUD
            const titleElem = document.getElementById('player-hud-title');
            if (titleElem) titleElem.textContent = videoName;
            const activePill = document.getElementById('player-active-video-name');
            if (activePill) activePill.textContent = videoName;
            const statusText = document.getElementById('player-status-text');
            if (statusText) statusText.textContent = 'Đang Phát Trực Tiếp';

            updatePlayerSource();

            // Bắt đầu telemetry polling
            if (streamTelemetryInterval) clearInterval(streamTelemetryInterval);
            streamTelemetryInterval = setInterval(pollStreamTelemetry, 500);
        }

        function updatePlayerSource() {
            if (!currentStreamVideo) return;
            const imgPlayer = document.getElementById('ai-live-stream-player');
            const rawPlayer = document.getElementById('raw-video-player');

            if (currentPlayerMode === 'ai') {
                if (rawPlayer) {
                    rawPlayer.pause();
                    rawPlayer.style.display = 'none';
                }
                if (imgPlayer) {
                    imgPlayer.style.display = 'block';
                    imgPlayer.src = `/api/stream/video?name=${encodeURIComponent(currentStreamVideo)}&token=${currentStreamToken}&skip=${currentFrameSkip}&overlay=1&conf=${currentConfidence}&imgsz=${currentImgsz}`;
                }
            } else {
                if (imgPlayer) {
                    imgPlayer.style.display = 'none';
                    imgPlayer.src = '';
                }
                if (rawPlayer) {
                    rawPlayer.style.display = 'block';
                    rawPlayer.src = `/videos/${encodeURIComponent(currentStreamVideo)}`;
                    rawPlayer.play().catch(e => console.log('Autoplay:', e));
                }
            }
        }

        function pauseLiveStream() {
            const rawPlayer = document.getElementById('raw-video-player');
            if (currentPlayerMode === 'raw' && rawPlayer) {
                rawPlayer.pause();
            } else {
                currentStreamToken = 0;
            }
            const statusText = document.getElementById('player-status-text');
            if (statusText) statusText.textContent = 'Tạm Dừng';
        }

        function resumeLiveStream() {
            if (!currentStreamVideo) {
                if (allPlayerVideos.length > 0) selectAndPlayVideo(allPlayerVideos[0].filename);
                return;
            }
            currentStreamToken = Date.now();
            updatePlayerSource();
            const statusText = document.getElementById('player-status-text');
            if (statusText) statusText.textContent = 'Đang Phát Trực Tiếp';
        }

        function restartLiveStream() {
            if (!currentStreamVideo) return;
            currentStreamToken = Date.now();
            const rawPlayer = document.getElementById('raw-video-player');
            if (currentPlayerMode === 'raw' && rawPlayer) {
                rawPlayer.currentTime = 0;
                rawPlayer.play().catch(e => {});
            } else {
                updatePlayerSource();
            }
        }

        function stopLiveStream() {
            currentStreamToken = 0;
            const imgPlayer = document.getElementById('ai-live-stream-player');
            if (imgPlayer) {
                imgPlayer.src = '';
                imgPlayer.style.display = 'none';
            }
            const rawPlayer = document.getElementById('raw-video-player');
            if (rawPlayer) {
                rawPlayer.pause();
                rawPlayer.style.display = 'none';

            const placeholder = document.getElementById('player-screen-placeholder');
            if (placeholder) placeholder.style.display = 'flex';

                streamTelemetryInterval = null;

        async function pollStreamTelemetry() {
            if (!currentStreamVideo || currentPlayerMode !== 'ai') return;
            try {
                const res = await fetch(`/api/stream/telemetry?video=${encodeURIComponent(currentStreamVideo)}`);
                if (!res.ok) return;

                if (!data || data.status !== 'streaming') return;

                if (fpsStat) fpsStat.textContent = `${data.fps || 0.0} FPS`;

                const violStat = document.getElementById('player-stat-viol');

                    if (violAlert) violAlert.style.display = 'flex';

                    chipsContainer.innerHTML = data.events.map(ev => {
                        let badgeClass = 'chip-normal';

                            badgeClass = 'chip-suspected';
                            icon = 'fa-clock';
                            statusLabel = 'Nghi vấn';

                            <div class="live-obj-chip ${badgeClass}">
                                <i class="fa-solid ${icon}"></i>

                                <span style="font-weight: 700;">${statusLabel}</span>

        async function startEvaluation(explicitMode = null, explicitVideo = null) {
            if (evalRunning) return;

            const modeSelect = document.getElementById('eval-mode-select');
            const videoSelect = document.getElementById('eval-video-select');
            
            let mode = explicitMode || (modeSelect ? modeSelect.value : 'batch');
            let videoName = explicitVideo || (videoSelect && mode === 'single' ? videoSelect.value : '');

            if (mode === 'single' && !videoName) {
                alert('Vui lòng chọn video để kiểm thử!');
                return;
            }

            setEvalRunningUI(true);
            evalLogOffset = 0;
            const consoleScreen = document.getElementById('eval-console-screen');
            if (consoleScreen) consoleScreen.textContent = `[${new Date().toLocaleTimeString()}] Bắt đầu tiến trình kiểm thử (${mode.toUpperCase()})...\n`;

            try {
                const res = await fetch('/api/eval/run', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ mode: mode, video: videoName })
                });
                const data = await res.json();
                if (!res.ok || !data.ok) {
                    alert('Lỗi: ' + (data.error || 'Không thể bắt đầu kiểm thử.'));
                    setEvalRunningUI(false);
                    return;
                }

                if (evalPollInterval) clearInterval(evalPollInterval);
                evalPollInterval = setInterval(pollEvalStatus, 600);
            } catch (e) {
                console.error('Lỗi gọi API /api/eval/run:', e);
                alert('Không thể kết nối đến máy chủ.');
                setEvalRunningUI(false);
            }
        }

        async function checkInitialEvalStatus() {
            try {
                const res = await fetch('/api/eval/status?since=0');
                if (!res.ok) return;
                const data = await res.json();
                if (data.status === 'running') {
                    setEvalRunningUI(true);
                    if (evalPollInterval) clearInterval(evalPollInterval);
                    evalPollInterval = setInterval(pollEvalStatus, 600);
                } else if (data.status === 'completed' && data.has_results) {
                    loadEvaluationResults();
                }
            } catch (e) {}
        }

        async function pollEvalStatus() {
            try {
                const res = await fetch(`/api/eval/status?since=${evalLogOffset}`);
                if (!res.ok) return;
                const data = await res.json();

                if (data.new_logs && data.new_logs.length > 0) {
                    const screen = document.getElementById('eval-console-screen');
                    if (screen) {
                        data.new_logs.forEach(l => {
                            screen.textContent += l + String.fromCharCode(10);
                        });
                        evalLogOffset = data.total_logs;
                        const autoscroll = document.getElementById('eval-autoscroll');
                        if (autoscroll && autoscroll.checked) {
                            screen.scrollTop = screen.scrollHeight;
                        }
                    }
                }

                const pct = Math.min(100, Math.max(0, data.percent || 0));
                const fill = document.getElementById('eval-progress-fill');
                if (fill) fill.style.width = pct + '%';
                const pctLabel = document.getElementById('eval-progress-pct');
                if (pctLabel) pctLabel.textContent = Math.round(pct) + '%';

                const metaLabel = document.getElementById('eval-progress-label');
                if (metaLabel) metaLabel.textContent = data.message || 'Đang thực thi...';

                const frameStat = document.getElementById('eval-stat-frame');
                if (frameStat) frameStat.textContent = `${data.current_frame} / ${data.total_frames}`;

                const fpsStat = document.getElementById('eval-stat-fps');
                if (fpsStat) fpsStat.textContent = `${data.fps || 0.0} FPS`;

                const timeStat = document.getElementById('eval-stat-time');
                if (timeStat) timeStat.textContent = `${data.elapsed_sec || 0.0}s`;

                if (data.status === 'completed') {
                    if (evalPollInterval) clearInterval(evalPollInterval);
                    evalPollInterval = null;
                    setEvalRunningUI(false);
                    updateStatusPill('completed');
                    showToast('Đã hoàn thành kiểm thử video thành công!', 'success');
                    loadEvaluationResults();
                } else if (data.status === 'error') {
                    if (evalPollInterval) clearInterval(evalPollInterval);
                    evalPollInterval = null;
                    setEvalRunningUI(false);
                    updateStatusPill('error');
                    alert('Lỗi kiểm thử: ' + (data.error || data.message));
                }
            } catch (e) {
                console.error('Lỗi polling:', e);
            }
        }

        function setEvalRunningUI(isRunning) {
            evalRunning = isRunning;
            const btnStart = document.getElementById('btn-start-eval');
            const iconStart = document.getElementById('btn-start-icon');
            const textStart = document.getElementById('btn-start-text');

                if (iconStart) iconStart.className = 'fa-solid fa-play';
                if (textStart) textStart.textContent = 'Bắt Đầu Chạy Kiểm Thử';
                if (dot) dot.style.opacity = '0.4';
            }
        }

        function updateStatusPill(status) {

                text.textContent = 'Có Lỗi Xảy Ra';
                pill.style.borderColor = 'rgba(239, 68, 68, 0.4)';

                pill.style.borderColor = 'var(--border)';
            }

        function renderEvaluationResults(data) {
            const container = document.getElementById('eval-results-container');
            if (!container) return;
            container.style.display = 'block';

            const isBatch = !!data.metrics_version_b_official;
            const meta = data.metadata || {};
            const tsLabel = document.getElementById('results-timestamp');
            if (tsLabel) {
                tsLabel.textContent = `Cập nhật lúc: ${meta.generated_at || new Date().toLocaleString()}`;

_FRAME_CACHE = {}


def get_roi_videos_list():
    """Quét và trả về danh sách tất cả các tệp video kèm trạng thái cấu hình ROI."""
    video_dir = os.path.abspath("data/videos")
    config_dir = os.path.abspath("configs")
    if not os.path.exists(video_dir):
        return []

    videos = []
    valid_exts = ('.mp4', '.avi', '.mov', '.mkv', '.flv', '.wmv')
    for fname in sorted(os.listdir(video_dir)):
        if fname.lower().endswith(valid_exts) and not fname.startswith('.'):
            fpath = os.path.join(video_dir, fname)
            try:
                size_mb = round(os.path.getsize(fpath) / (1024 * 1024), 2)
            except Exception:
                size_mb = 0

            w, h, fps, total_frames = 1280, 720, 24.0, 0
            try:
                cap = cv2.VideoCapture(fpath)
                vw = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
                vh = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
                vfps = round(cap.get(cv2.CAP_PROP_FPS), 1)
                vframes = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
                cap.release()
                if vw > 0 and vh > 0:
                    w, h, fps, total_frames = vw, vh, vfps, vframes
            except Exception:
                pass

            base_name = os.path.splitext(fname)[0]
            clean_name = base_name.replace(" ", "_").replace(",", "").replace("-", "_")

            candidate_files = [
                f"roi_{clean_name}.json",
                f"roi_{base_name}.json"
            ]
            cfg_file = None
            for c in candidate_files:
                p = os.path.join(config_dir, c)
                if os.path.exists(p):
                    cfg_file = c
                    break

            has_roi = False
            regions_count = 0
            points_count = 0
            created_at = ""
            if cfg_file:
                try:
                    with open(os.path.join(config_dir, cfg_file), "r", encoding="utf-8") as cf:
                        cdata = json.load(cf)
                        polys = cdata.get("polygons", [])
                        if not polys and "polygon_points" in cdata and len(cdata["polygon_points"]) >= 3:
                            polys = [cdata["polygon_points"]]
                        regions_count = len(polys)
                        points_count = sum(len(p) for p in polys)
                        has_roi = regions_count > 0
                        created_at = cdata.get("created_at", "")
                except Exception:
                    pass

            videos.append({
                "filename": fname,
                "clean_id": clean_name,
                "size_mb": size_mb,
                "width": w,
                "height": h,
                "fps": fps,
                "total_frames": total_frames,
                "has_roi": has_roi,
                "config_filename": cfg_file or f"roi_{clean_name}.json",
                "regions_count": regions_count,
                "points_count": points_count,
                "created_at": created_at
            })

    return videos


def extract_video_frame(video_name: str) -> bytes:
    """Trích xuất khung hình đầu tiên (frame 0) của video và trả về bytes JPEG."""
    video_name = os.path.basename(video_name)
    video_path = os.path.abspath(os.path.join("data", "videos", video_name))

    data_dir = os.path.abspath("data/videos")
    if os.path.commonpath([data_dir, video_path]) != data_dir or not os.path.exists(video_path):
        return None

    if video_path in _FRAME_CACHE:
        return _FRAME_CACHE[video_path]

    try:
        cap = cv2.VideoCapture(video_path)
        ret, frame = cap.read()
        cap.release()
        if not ret or frame is None:
            return None

        success, buf = cv2.imencode('.jpg', frame, [cv2.IMWRITE_JPEG_QUALITY, 85])
        if success:
            jpeg_bytes = buf.tobytes()
            _FRAME_CACHE[video_path] = jpeg_bytes
            return jpeg_bytes
    except Exception as e:
        print(f"[ERROR] Không thể đọc frame từ video {video_name}: {e}")

    return None

def save_roi_config(data: dict) -> dict:
    """Lưu danh sách đa giác vỉa hè ra tệp JSON chuẩn ROIManager."""

        return {"ok": False, "error": "Tên video không hợp lệ"}

    clean_name = base_name.replace(" ", "_").replace(",", "").replace("-", "_")
    config_dir = os.path.abspath("configs")
    os.makedirs(config_dir, exist_ok=True)

    raw_polygons = data.get("polygons", [])
    valid_polygons = []
    for poly in raw_polygons:
        if isinstance(poly, list) and len(poly) >= 3:
            valid_polygons.append([[int(pt[0]), int(pt[1])] for pt in poly if len(pt) >= 2])

    res = data.get("resolution", {})
    width = int(res.get("width", 1280)) if isinstance(res, dict) else 1280
    height = int(res.get("height", 720)) if isinstance(res, dict) else 720

    out_filename = f"roi_{clean_name}.json"
    out_path = os.path.join(config_dir, out_filename)

    total_pts = sum(len(p) for p in valid_polygons)
    save_data = {
        "camera_id": base_name,
        "description": "Vung via he co dinh (Ho tro Multi-ROI: Via he trai, phai)",
        "resolution": {"width": width, "height": height},
        "total_regions": len(valid_polygons),

        "polygon_points": valid_polygons[0] if valid_polygons else [],

_EVAL_LOCK = threading.Lock()
_EVAL_STATE = {
    "status": "idle",       # "idle", "running", "completed", "error"
    "mode": "batch",        # "batch" or "single"
    "video_name": "",
    "current_frame": 0,
    "total_frames": 0,
    "fps": 0.0,
    "percent": 0.0,
    "elapsed_sec": 0.0,
    "message": "Sẵn sàng",
    "logs": [],
    "results": None,
    "error": None
}


class WebEvalLogger:
    """Thu thập dòng log streaming từ pipeline đánh giá đưa vào bộ nhớ đệm web."""
    def write(self, msg):
        if not msg:
            return
        try:
            sys.stdout.write(msg)
            sys.stdout.flush()
        except Exception:
            pass
        
        lines = msg.splitlines(keepends=False)
        with _EVAL_LOCK:
            for l in lines:
                if l.strip():
                    _EVAL_STATE["logs"].append(l)
            if len(_EVAL_STATE["logs"]) > 2500:
                _EVAL_STATE["logs"] = _EVAL_STATE["logs"][-2000:]

    def flush(self):
        try:
            sys.stdout.flush()
        except Exception:
            pass


def _eval_worker_func(mode: str, video_name: str):
    """Hàm chạy ngầm đánh giá video trên thread riêng biệt, không block web server."""
    from evaluation import evaluate_videos as ev
    logger = WebEvalLogger()
    t_start = time.time()

    def _batch_cb(pct, vid_label, f_idx, f_total, fps_val):
        with _EVAL_LOCK:
            _EVAL_STATE["percent"] = pct
            _EVAL_STATE["current_frame"] = f_idx
            _EVAL_STATE["total_frames"] = f_total
            _EVAL_STATE["fps"] = fps_val
            _EVAL_STATE["elapsed_sec"] = time.time() - t_start
            _EVAL_STATE["message"] = f"Đang đánh giá {vid_label}: {f_idx}/{f_total} frames ({fps_val} FPS)"

    def _single_cb(f_idx, f_total, fps_val):
        with _EVAL_LOCK:
            pct = int((f_idx / max(1, f_total)) * 100)
            _EVAL_STATE["percent"] = pct
            _EVAL_STATE["current_frame"] = f_idx
            _EVAL_STATE["total_frames"] = f_total
            _EVAL_STATE["fps"] = fps_val
            _EVAL_STATE["elapsed_sec"] = time.time() - t_start
            _EVAL_STATE["message"] = f"Đang xử lý {video_name}: {f_idx}/{f_total} frames ({fps_val} FPS)"

    try:
        if mode == "batch":
            with _EVAL_LOCK:
                _EVAL_STATE["message"] = "Đang chạy đánh giá toàn diện 3 video nghiên cứu..."
            res = ev.run_full_evaluation(progress_callback=_batch_cb, custom_logger=logger)
            with _EVAL_LOCK:
                _EVAL_STATE["results"] = res
                _EVAL_STATE["status"] = "completed"
                _EVAL_STATE["percent"] = 100.0
                _EVAL_STATE["elapsed_sec"] = time.time() - t_start
                _EVAL_STATE["message"] = "Đã hoàn thành đánh giá toàn diện 3 video thành công!"
                _EVAL_STATE["logs"].append(f"[{time.strftime('%H:%M:%S')}] HOÀN THÀNH ĐÁNH GIÁ TOÀN DIỆN 3 VIDEO.")
        else:
            with _EVAL_LOCK:
                _EVAL_STATE["message"] = f"Đang chạy kiểm thử video {video_name}..."
            res = ev.run_single_video_eval_named(video_name, progress_callback=_single_cb, custom_logger=logger)
            with _EVAL_LOCK:
                if "error" in res:
                    _EVAL_STATE["status"] = "error"
                    _EVAL_STATE["error"] = res["error"]
                    _EVAL_STATE["message"] = res["error"]
                    _EVAL_STATE["logs"].append(f"[ERROR] {res['error']}")
                else:
                    _EVAL_STATE["results"] = res
                    _EVAL_STATE["status"] = "completed"
                    _EVAL_STATE["percent"] = 100.0

    except Exception as e:

        tb = traceback.format_exc()

            _EVAL_STATE["status"] = "error"

    def serve_video_file(self, file_path):
        """Phục vụ file video mp4 hỗ trợ HTTP 206 Partial Content (Range header) cho HTML5 player."""
        if not os.path.isfile(file_path):
            self.send_response(404)
            self.end_headers()
            return

        file_size = os.path.getsize(file_path)
        range_header = self.headers.get("Range")
        if range_header:
            range_match = re.match(r"bytes=(\d+)-(\d*)", range_header)
            if range_match:
                start = int(range_match.group(1))
                end = int(range_match.group(2)) if range_match.group(2) else file_size - 1
                length = end - start + 1
                self.send_response(206)
                self.send_header("Content-Type", "video/mp4")
                self.send_header("Content-Range", f"bytes {start}-{end}/{file_size}")
                self.send_header("Content-Length", str(length))
                self.send_header("Accept-Ranges", "bytes")
                self.send_header("Cache-Control", "no-cache")
                self.end_headers()
                try:
                    with open(file_path, "rb") as f:
                        f.seek(start)
                        self.wfile.write(f.read(length))
                except (BrokenPipeError, ConnectionResetError, OSError):
                    pass
                return

        self.send_response(200)
        self.send_header("Content-Type", "video/mp4")
        self.send_header("Content-Length", str(file_size))
        self.send_header("Accept-Ranges", "bytes")
        self.end_headers()
        try:
            with open(file_path, "rb") as f:
                while chunk := f.read(65536):
                    self.wfile.write(chunk)
        except (BrokenPipeError, ConnectionResetError, OSError):
            pass

    def stream_ai_video(self, video_name, token, frame_skip=1, overlay=True, conf=0.15, imgsz=960):
        """Truyền phát trực tiếp luồng video AI (MJPEG Stream) kèm nhận diện YOLO & bám vết ByteTrack."""
        video_name = os.path.basename(video_name)
        video_path = os.path.abspath(os.path.join("data", "videos", video_name))
        data_dir = os.path.abspath("data/videos")

        if os.path.commonpath([data_dir, video_path]) != data_dir or not os.path.exists(video_path):
            self.send_response(404)
            self.end_headers()
            return

        with _STREAM_LOCK:
            _ACTIVE_STREAM_TOKENS[video_name] = token
            _STREAM_TELEMETRY[video_name] = {
                "status": "streaming",
                "video": video_name,
                "current_frame": 0,
                "total_frames": 0,
                "fps": 0.0,
                "objects_count": 0,
                "violations_count": 0,
                "events": []
            }

        self.send_response(200)
        self.send_header("Content-Type", "multipart/x-mixed-replace; boundary=frame")
        self.send_header("Cache-Control", "no-cache, no-store, must-revalidate")
        self.send_header("Pragma", "no-cache")
        self.send_header("Expires", "0")
        self.send_header("Connection", "close")
        self.end_headers()

        cap = cv2.VideoCapture(video_path)
        fps_source = cap.get(cv2.CAP_PROP_FPS) or 24.0
        total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT)) or 0
        frame_delay = 1.0 / max(10.0, fps_source)

        from tracking import ObjectTracker, TemporalVerifier, TrackState
        from roi import ROIManager
        from violation import ViolationChecker
        from utils import FPSTracker, draw_fps_badge, draw_detection
        from evidence import EvidenceSaver

        roi_info = get_roi_config(video_name)
        config_path = os.path.join("configs", roi_info.get("config_path", ""))
        roi_manager = ROIManager(config_path if os.path.exists(config_path) else "configs/roi_camera1.json")
        camera_clean_id = roi_manager.camera_id or os.path.splitext(video_name)[0]
        checker = ViolationChecker(threshold=0.30)
        min_avg_c = max(0.25, min(0.40, conf * 1.6))
        verifier = TemporalVerifier(
            confirm_frames=20,
            confirm_seconds=3.0,
            source_fps=fps_source,
            min_avg_confidence=min_avg_c
        )
        fps_tracker = FPSTracker(alpha=0.15)
        tracker = get_shared_tracker(conf_threshold=conf, imgsz=imgsz)
        with _TRACKER_INFER_LOCK:
            tracker.reset()
        evidence_saver = EvidenceSaver(base_dir="evidence", blur_faces=True)

        frame_idx = 0
        roi_validated = False
        last_results = []
        confirmed_ids = set()

        try:
            while True:
                with _STREAM_LOCK:
                    if _ACTIVE_STREAM_TOKENS.get(video_name) != token:
                        break

                t_start = time.perf_counter()
                ret, frame = cap.read()
                if not ret or frame is None:
                    # Tua lại video từ đầu để tiếp tục phát mượt mà
                    cap.set(cv2.CAP_PROP_POS_FRAMES, 0)
                    frame_idx = 0
                    verifier.reset()
                    with _TRACKER_INFER_LOCK:
                        tracker.reset()
                    ret, frame = cap.read()
                    if not ret or frame is None:
                        break

                frame_idx += 1
                fh, fw = frame.shape[:2]
                if not roi_validated:
                    roi_manager.validate_resolution(fw, fh)
                    roi_validated = True

                clean_frame = frame.copy()
                curr_fps = fps_tracker.update()
                is_infer = (frame_skip <= 0) or (frame_idx % (frame_skip + 1) == 1)

                if is_infer:
                    with _TRACKER_INFER_LOCK:
                        tracked_objs = tracker.track(frame, conf=conf, imgsz=imgsz)
                    sim_ts = frame_idx / fps_source
                    last_results = verifier.process_frame(
                        tracked_objs, checker, roi_manager.polygons, current_timestamp=sim_ts
                    )

                    events = []
                    viol_count = 0
                    for obj in last_results:
                        tid = obj["track_id"]
                        stat = obj["temporal_status"]
                        sp = obj["spatial_res"]
                        box = obj["box"]
                        if stat == TrackState.CONFIRMED:
                            viol_count += 1
                            confirmed_ids.add(tid)
                            events.append({"track_id": tid, "status": "CONFIRMED", "overlap": sp["overlap_pct"]})

                            # Tự động lưu bằng chứng ảnh & ghi nhận hồ sơ vi phạm vào CSDL (phục vụ duyệt phạt nguội)
                            if obj.get("is_new_alert", False):
                                if not db.is_duplicate_violation(camera_clean_id, box, time_window_minutes=5):
                                    sw_idx = sp.get("sidewalk_index", 1)
                                    ev_res = evidence_saver.save_evidence(
                                        clean_frame, box, track_id=tid, camera_id=camera_clean_id, overlap_pct=sp["overlap_pct"]
                                    )
                                    rec_id = db.insert_violation(
                                        camera_id=camera_clean_id,
                                        track_id=tid,
                                        timestamp=ev_res["timestamp"],
                                        confidence=obj["conf"],
                                        overlap_pct=sp["overlap_pct"],
                                        full_image_path=ev_res["full_path"],
                                        crop_image_path=ev_res.get("crop_path", ""),
                                        bbox=box,
                                        sidewalk_index=sw_idx
                                    )
                                    print(f"   [WEB LIVE AI] Đã lưu bằng chứng & hồ sơ vi phạm #{rec_id} cho Track #{tid} ({camera_clean_id})")
                        elif stat == TrackState.SUSPECTED:
                            events.append({"track_id": tid, "status": "SUSPECTED", "overlap": sp["overlap_pct"]})
                        else:
                            events.append({"track_id": tid, "status": "NORMAL", "overlap": sp["overlap_pct"]})

                    with _STREAM_LOCK:
                        _STREAM_TELEMETRY[video_name] = {
                            "status": "streaming",
                            "video": video_name,
                            "current_frame": frame_idx,
                            "total_frames": total_frames,
                            "fps": round(curr_fps, 1),
                            "objects_count": len(last_results),
                            "violations_count": len(confirmed_ids),
                            "events": events[:8]
                        }

                if overlay:
                    roi_manager.draw_overlay(frame, alpha=0.25)
                    for obj in last_results:
                        box = obj["box"]
                        tid = obj["track_id"]
                        stat = obj["temporal_status"]
                        sp = obj["spatial_res"]
                        ov_pct = sp["overlap_pct"]
                        if stat == TrackState.CONFIRMED:
                            lbl = f"ID #{tid} VI PHAM {ov_pct:.0f}%"
                        elif stat == TrackState.SUSPECTED:
                            lbl = f"ID #{tid} NGHI VAN {ov_pct:.0f}%"
                        else:
                            lbl = f"ID #{tid} HOP LE {ov_pct:.0f}%"
                        draw_detection(frame, box, lbl, is_violation=(stat == TrackState.CONFIRMED), track_id=tid, status=stat)

                    draw_fps_badge(frame, curr_fps)
                    hud_title = f"{video_name} | Khung {frame_idx}/{total_frames} | Vi pham: {len(confirmed_ids)}"
                    cv2.putText(frame, hud_title, (20, fh - 20), cv2.FONT_HERSHEY_SIMPLEX, 0.65, (0, 0, 0), 4)
                    cv2.putText(frame, hud_title, (20, fh - 20), cv2.FONT_HERSHEY_SIMPLEX, 0.65, (56, 189, 248), 2)

                if fw > 960:
                    scale = 960 / fw
                    render_frame = cv2.resize(frame, (960, int(fh * scale)), interpolation=cv2.INTER_LINEAR)
                else:
                    render_frame = frame

                _, jpeg = cv2.imencode('.jpg', render_frame, [cv2.IMWRITE_JPEG_QUALITY, 75])
                frame_bytes = jpeg.tobytes()

                chunk = (
                    b"--frame\r\n"
                    b"Content-Type: image/jpeg\r\n"
                    b"Content-Length: " + str(len(frame_bytes)).encode() + b"\r\n\r\n"
                    + frame_bytes + b"\r\n"
                )
                self.wfile.write(chunk)
                self.wfile.flush()

                elapsed = time.perf_counter() - t_start
                sleep_time = max(0.002, frame_delay - elapsed)
                time.sleep(sleep_time)

        except (BrokenPipeError, ConnectionResetError, ConnectionAbortedError, OSError):
            pass
        except Exception as e:
            import traceback
            print(f"[STREAM ERROR] {e}\n{traceback.format_exc()}")
        finally:
            cap.release()
            with _STREAM_LOCK:
                if _ACTIVE_STREAM_TOKENS.get(video_name) == token:
                    _STREAM_TELEMETRY[video_name]["status"] = "stopped"

    def do_POST(self):
        """Xử lý cập nhật trạng thái vi phạm (Human-in-the-Loop)."""
        parsed = urllib.parse.urlparse(self.path)
        path = parsed.path

        # POST /api/violations/{id}/status
        match = re.match(r'/api/violations/(\d+)/status', path)
        if match:
            # Kiểm tra bảo mật: chỉ chấp nhận từ localhost
            client_host = self.client_address[0] if self.client_address else ""
            if client_host not in ("127.0.0.1", "::1", "localhost"):
                self.send_response(403)
                self.end_headers()
                self.wfile.write(b"403 Forbidden: Status updates require local access")
                return

            violation_id = int(match.group(1))
            content_length = int(self.headers.get('Content-Length', 0))
            body = self.rfile.read(content_length).decode('utf-8')
            try:
                data = json.loads(body)
                new_status = data.get('status', '')
                reviewed_by = data.get('reviewed_by', None)
                dismiss_reason = data.get('dismiss_reason', None)

                success = db.update_violation_status(
                    violation_id=violation_id,
                    new_status=new_status,
                    reviewed_by=reviewed_by,
                    dismiss_reason=dismiss_reason
                )
                self.send_response(200 if success else 400)
                self.send_header("Content-Type", "application/json; charset=utf-8")
                self.end_headers()
                self.wfile.write(json.dumps({"ok": success}).encode("utf-8"))
            except Exception as e:
                self.send_response(500)
                self.send_header("Content-Type", "application/json")
                self.end_headers()
                self.wfile.write(json.dumps({"error": str(e)}).encode("utf-8"))

        elif path == "/api/roi/save":
            client_host = self.client_address[0] if self.client_address else ""
            if client_host not in ("127.0.0.1", "::1", "localhost"):
                self.send_response(403)
                self.end_headers()
                self.wfile.write(b"403 Forbidden: Saving ROI requires local access")
                return

            content_length = int(self.headers.get('Content-Length', 0))
            body = self.rfile.read(content_length).decode('utf-8')
            try:
                data = json.loads(body)
                res = save_roi_config(data)
                self.send_response(200 if res.get("ok") else 400)
                self.send_header("Content-Type", "application/json; charset=utf-8")
                self.end_headers()
                self.wfile.write(json.dumps(res, ensure_ascii=False).encode("utf-8"))
            except Exception as e:
                self.send_response(500)
                self.send_header("Content-Type", "application/json; charset=utf-8")
                self.end_headers()
                self.wfile.write(json.dumps({"ok": False, "error": str(e)}, ensure_ascii=False).encode("utf-8"))
        elif path == "/api/eval/run":
            client_host = self.client_address[0] if self.client_address else ""
            if client_host not in ("127.0.0.1", "::1", "localhost"):
                self.send_response(403)
                self.end_headers()
                self.wfile.write(b"403 Forbidden: Evaluation requires local access")
                return

            content_length = int(self.headers.get('Content-Length', 0))
            body = self.rfile.read(content_length).decode('utf-8')
            try:
                data = json.loads(body) if body else {}
                mode = data.get("mode", "batch")
                video_name = data.get("video", "")

                with _EVAL_LOCK:
                    if _EVAL_STATE["status"] == "running":
                        self.send_response(409)
                        self.send_header("Content-Type", "application/json; charset=utf-8")
                        self.end_headers()
                        self.wfile.write(json.dumps({"ok": False, "error": "Tiến trình kiểm thử đang chạy dở. Vui lòng đợi!"}, ensure_ascii=False).encode("utf-8"))
                        return
                    _EVAL_STATE["status"] = "running"
                    _EVAL_STATE["mode"] = mode
                    _EVAL_STATE["video_name"] = video_name
                    _EVAL_STATE["current_frame"] = 0
                    _EVAL_STATE["total_frames"] = 0
                    _EVAL_STATE["fps"] = 0.0
                    _EVAL_STATE["percent"] = 0.0
                    _EVAL_STATE["elapsed_sec"] = 0.0
                    _EVAL_STATE["message"] = f"Khởi động kiểm thử ({mode.upper()})..."
                    _EVAL_STATE["logs"] = [f"[{time.strftime('%H:%M:%S')}] Khởi động kiểm thử video (Chế độ: {mode.upper()})..."]
                    _EVAL_STATE["error"] = None

                th = threading.Thread(target=_eval_worker_func, args=(mode, video_name), daemon=True)
                th.start()
                self.send_response(200)
                self.send_header("Content-Type", "application/json; charset=utf-8")
                self.end_headers()
                self.wfile.write(json.dumps({"ok": True, "status": "started"}, ensure_ascii=False).encode("utf-8"))
            except Exception as e:
                self.send_response(500)
                self.send_header("Content-Type", "application/json; charset=utf-8")
                self.end_headers()
                self.wfile.write(json.dumps({"ok": False, "error": str(e)}, ensure_ascii=False).encode("utf-8"))
        else:
            self.send_response(404)
            self.end_headers()

"""
Test suite cho module tracking/verifier.py (Temporal Verification & Cooldown Logic)

v2.1: Sửa để truyền current_timestamp mô phỏng vào process_frame,
      đảm bảo confirm_seconds=0.5 có thể đạt được trong test nhanh.
"""

import os
import sys

if hasattr(sys.stdout, 'reconfigure'):
    sys.stdout.reconfigure(encoding='utf-8')

sys.path.insert(0, os.path.abspath("."))

from tracking import TemporalVerifier, TrackState
from violation import ViolationChecker


def test_temporal_verifier():
    print("=" * 65)
    print("KIEM THU BO LOC THOI GIAN VA COOLDOWN (TASK 22 - TEMPORAL VERIFIER)")
    print("=" * 65)

    checker = ViolationChecker(threshold=0.30)

    # confirm_frames=5, confirm_seconds=0.5
    # Dung fps_sim=8: moi frame cach 0.125s => sau 4 frames = 0.5s dat nguong giay
    verifier = TemporalVerifier(
        confirm_frames=5,
        confirm_seconds=0.5,
        cooldown_seconds=10.0,
        max_missing_frames=5,
    )

    # Da giac via he gia lap
    test_poly = [[[100, 100], [500, 100], [500, 500], [100, 500]]]
    box_violating = (200, 200, 300, 400)
    fps_sim = 8.0   # moi frame cach 125ms => frame thu 5 (index 4) cach 0.5s

    # -----------------------------------------------------------------------
    # 1. Kiem tra doi tuong dung yen lan chiem via he (ID #1)
    # -----------------------------------------------------------------------
    print("\n1. Kiem tra doi tuong dung yen lan chiem via he (ID #1):")
    alert_triggered_count = 0
    base_ts = 1000.0   # timestamp co dinh de co the lap lai duoc

    for frame_idx in range(1, 8):
        tracked_objs = [{
            "track_id": 1,
            "box": box_violating,
            "conf": 0.85,
            "class_id": 0,
            "class_name": "signboard"
        }]
        ts = base_ts + (frame_idx - 1) / fps_sim
        results = verifier.process_frame(tracked_objs, checker, test_poly, current_timestamp=ts)
        res = results[0]

        status = res["temporal_status"]
        count = res["violation_frames"]
        is_alert = res["is_new_alert"]
        if is_alert:
            alert_triggered_count += 1

        elapsed = ts - base_ts
        print(f"  Frame #{frame_idx} (t={elapsed:.3f}s): Count={count} | Status={status} | Alert={is_alert}")

        if frame_idx < 5:
            assert status == TrackState.SUSPECTED, f"Frame {frame_idx} phai la SUSPECTED!"
            assert not is_alert, "Chua du 5 frames khong duoc phep phat canh bao!"
        elif frame_idx == 5:
            assert status == TrackState.CONFIRMED, "Frame 5 phai nang cap thanh CONFIRMED!"
            assert is_alert, "Frame 5 vua dat nguong phai kich hoat New Alert = True!"
        elif frame_idx > 5:
            assert status == TrackState.CONFIRMED, "Van tiep tuc la CONFIRMED!"
            assert not is_alert, "Frame sau do phai Cooldown, khong duoc spam Alert!"

    assert alert_triggered_count == 1, "Co che Cooldown that bai, canh bao bi lap lai!"
    print("  => [PASS] Da xac nhan vi pham dung sau 5 frames lien tiep va Cooldown thanh cong!")

    # -----------------------------------------------------------------------
    # 2. Kiem tra doi tuong thoang qua (ID #2)
    # -----------------------------------------------------------------------
    print("\n2. Kiem tra doi tuong thoang qua / Nguoi be bien di ngang (ID #2):")
    base_ts2 = base_ts + 100.0   # thoi diem moi, tranh dinh cooldown
    transient_obj = [{
        "track_id": 2, "box": box_violating, "conf": 0.80,
        "class_id": 0, "class_name": "signboard"
    }]
    res_f1 = verifier.process_frame(transient_obj, checker, test_poly,
                                    current_timestamp=base_ts2)[0]
    res_f2 = verifier.process_frame(transient_obj, checker, test_poly,
                                    current_timestamp=base_ts2 + 1 / fps_sim)[0]

    box_legal = (600, 600, 700, 700)
    legal_obj = [{"track_id": 2, "box": box_legal, "conf": 0.80,
                  "class_id": 0, "class_name": "signboard"}]
    res_f3 = verifier.process_frame(legal_obj, checker, test_poly,
                                    current_timestamp=base_ts2 + 2 / fps_sim)[0]

    print(f"  Frame #1: Status={res_f1['temporal_status']} (Count:{res_f1['violation_frames']})")
    print(f"  Frame #2: Status={res_f2['temporal_status']} (Count:{res_f2['violation_frames']})")
    print(f"  Frame #3 (roi di): Status={res_f3['temporal_status']} (Count:{res_f3['violation_frames']})")

    assert res_f1["temporal_status"] == TrackState.SUSPECTED
    assert res_f2["temporal_status"] == TrackState.SUSPECTED
    assert res_f3["temporal_status"] == TrackState.SUSPECTED
    assert not res_f1["is_new_alert"] and not res_f2["is_new_alert"] and not res_f3["is_new_alert"]
    print("  => [PASS] Loai bo 100% canh bao gia cho doi tuong thoang qua!")

    # -----------------------------------------------------------------------
    # 3. Kiem tra co che Reset dung sai khi doi tuong khong vi pham
    # -----------------------------------------------------------------------
    print("\n3. Kiem tra co che Reset dung sai (Tolerance Reset):")
    res_f4 = verifier.process_frame(legal_obj, checker, test_poly,
                                    current_timestamp=base_ts2 + 3 / fps_sim)[0]
    res_f5 = verifier.process_frame(legal_obj, checker, test_poly,
                                    current_timestamp=base_ts2 + 4 / fps_sim)[0]
    print(f"  Frame #4 (Khong vi pham lan 2): Count={res_f4['violation_frames']} | Status={res_f4['temporal_status']}")
    print(f"  Frame #5 (Khong vi pham lan 3): Count={res_f5['violation_frames']} | Status={res_f5['temporal_status']}")
    assert res_f5["violation_frames"] == 0 and res_f5["temporal_status"] == TrackState.NORMAL
    print("  => [PASS] Da reset hoan toan bo dem ve 0 khi khong vi pham qua thoi gian dung sai!")

    # -----------------------------------------------------------------------
    # 4. Kiem tra Occlusion Decay (ID #3)
    # -----------------------------------------------------------------------
    print("\n4. Kiem tra xu ly khi doi tuong bi che khuat (Occlusion Decay):")
    base_ts4 = base_ts2 + 200.0
    obj3 = [{"track_id": 3, "box": box_violating, "conf": 0.85,
             "class_id": 0, "class_name": "signboard"}]
    for i in range(4):
        verifier.process_frame(obj3, checker, test_poly,
                               current_timestamp=base_ts4 + i / fps_sim)
    assert verifier.track_records[3]["count"] == 4

    # Doi tuong bien mat 4 frames
    for i in range(4):
        verifier.process_frame([], checker, test_poly,
                               current_timestamp=base_ts4 + (4 + i) / fps_sim)
    print("  => [PASS] Occlusion decay hoat dong!")

    # -----------------------------------------------------------------------
    # 5. Kiem tra sliding window stationary (doi tuong di dong - ID #4)
    # -----------------------------------------------------------------------
    print("\n5. Kiem tra doi tuong di chuyen lien tuc (Nguoi di bo mang bien):")
    base_ts5 = base_ts4 + 200.0
    res_moving = None
    for f in range(6):
        moving_box = (150 + f * 20, 200, 250 + f * 20, 400)
        obj_moving = [{"track_id": 4, "box": moving_box, "conf": 0.85,

        res_moving = verifier.process_frame(obj_moving, checker, test_poly,
                                            current_timestamp=base_ts5 + f / fps_sim)[0]

    print(f"  ID #4 sau 6 frames di chuyen: Status={res_moving['temporal_status']} | Alert={res_moving['is_new_alert']}")
    assert res_moving["temporal_status"] == TrackState.SUSPECTED, \
        "Doi tuong di chuyen khong duoc nang cap len CONFIRMED!"
    assert not res_moving["is_new_alert"], "Doi tuong di chuyen khong duoc phat canh bao!"
    print("  => [PASS] Loai bo thanh cong doi tuong di dong mang bien luot qua via he!")

    # -----------------------------------------------------------------------
    # 6. Kiem tra Spatial Deduplication khi ByteTrack doi ID
    # -----------------------------------------------------------------------
    print("\n6. Kiem tra khu trung lap khong gian khi ByteTrack doi ID (Spatial Deduplication):")
    verifier.reset()
    base_ts6 = 5000.0
    obj5 = [{"track_id": 5, "box": box_violating, "conf": 0.85,
             "class_id": 0, "class_name": "signboard"}]
    for i in range(5):
        res5 = verifier.process_frame(obj5, checker, test_poly,
                                      current_timestamp=base_ts6 + i / fps_sim)[0]
    assert res5["is_new_alert"], "ID #5 phai phat canh bao dau tien!"

    # Mat dau va sinh ID moi #6 tai cung vi tri
    obj6 = [{"track_id": 6, "box": (202, 201, 302, 401), "conf": 0.85,
             "class_id": 0, "class_name": "signboard"}]
    for i in range(5):
        res6 = verifier.process_frame(obj6, checker, test_poly,
                                      current_timestamp=base_ts6 + (5 + i) / fps_sim)[0]
    print(f"  ID #6 (ID moi tai vi tri cu trong cooldown): Status={res6['temporal_status']} | Alert={res6['is_new_alert']}")
    assert res6["temporal_status"] == TrackState.CONFIRMED
    assert not res6["is_new_alert"], "ID moi cung vi tri trong Cooldown khong duoc phat canh bao trung!"
    print("  => [PASS] Khu trung lap canh bao theo vi tri khong gian thanh cong 100%!")

    print("\n" + "=" * 65)
    print("[THANH CONG] 100% UNIT TEST BO LOC THOI GIAN (TASK 22) DAT CHUAN TUYET DOI!")
    print("=" * 65)
    return True


if __name__ == "__main__":

def get_default_config_for_source(source_path: str) -> str:
    """Tự động sinh đường dẫn file JSON ROI chuẩn tương ứng với tên file nguồn."""
    base_name = os.path.splitext(os.path.basename(source_path))[0]

def mouse_handler(event, x, y, flags, param):
    """Bắt sự kiện click chuột để thêm đỉnh hoặc xóa đỉnh của đa giác."""
    global current_polygon
    
    # Click chuột trái: Thêm 1 đỉnh mới cho đa giác đang vẽ
    if event == cv2.EVENT_LBUTTONDOWN:
        current_polygon.append([x, y])
        print(f"[ĐỈNH MỚI] Vỉa hè #{len(all_polygons)+1} - Đỉnh #{len(current_polygon)}: ({x}, {y})")
        
    # Click chuột phải: Xóa đỉnh gần nhất (Undo)
    elif event == cv2.EVENT_RBUTTONDOWN:
        if current_polygon:
            removed = current_polygon.pop()
            print(f"[UNDO] Đã xóa đỉnh: {removed}. Còn lại: {len(current_polygon)} đỉnh.")
        elif all_polygons:
            current_polygon = all_polygons.pop()
            print(f"[UNDO] Mở lại Vỉa hè #{len(all_polygons)+1} để sửa tiếp.")


def draw_multi_roi_overlay(frame, completed_polys, active_pts):
    """
    Vẽ tất cả các đa giác vỉa hè (đã chốt và đang vẽ) với lớp phủ bán trong suốt.
    """
    vis_frame = frame.copy()
    overlay = vis_frame.copy()

    # 1. Vẽ các vỉa hè ĐÃ CHỐT trước đó (Xanh cyan / Vàng)
    for idx, poly in enumerate(completed_polys):
        if len(poly) >= 3:
            pts_arr = np.array(poly, np.int32).reshape((-1, 1, 2))
            cv2.fillPoly(overlay, [pts_arr], (0, 200, 255))
            cv2.polylines(vis_frame, [pts_arr], isClosed=True, color=(0, 255, 255), thickness=2)
            first_pt = tuple(poly[0])
            cv2.putText(vis_frame, f"[VIA HE #{idx+1}]", (first_pt[0] + 5, max(25, first_pt[1] - 8)),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.55, (0, 255, 255), 2, cv2.LINE_AA)

    # 2. Vẽ vỉa hè ĐANG CHẤM DỞ
    if len(active_pts) > 0:
        for idx, (px, py) in enumerate(active_pts):
            cv2.circle(vis_frame, (px, py), 6, (0, 0, 255), -1)      # Chấm đỏ
            cv2.circle(vis_frame, (px, py), 7, (255, 255, 255), 2)  # Viền trắng
            cv2.putText(vis_frame, f"P{idx+1}", (px + 8, py - 5),
                        cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 255), 1, cv2.LINE_AA)

        if len(active_pts) >= 3:
            pts_arr = np.array(active_pts, np.int32).reshape((-1, 1, 2))
            cv2.fillPoly(overlay, [pts_arr], (255, 180, 0))
            cv2.polylines(vis_frame, [pts_arr], isClosed=True, color=(0, 200, 255), thickness=2)
        elif len(active_pts) == 2:
            cv2.line(vis_frame, tuple(active_pts[0]), tuple(active_pts[1]), (0, 200, 255), 2)

    cv2.addWeighted(overlay, 0.35, vis_frame, 0.65, 0, vis_frame)
    return vis_frame


def save_roi_config(polygons_list, width, height, camera_id="camera_01", target_path=None):
    """Lưu danh sách tất cả các đa giác vỉa hè ra file JSON chuẩn."""

    out_file = target_path if target_path else CONFIG_FILE

        "resolution": {"width": width, "height": height},
        "total_regions": len(polygons_list),
        "total_points": total_pts,
        "polygons": polygons_list,
        "polygon_points": polygons_list[0] if polygons_list else [],
        "created_at": time.strftime("%Y-%m-%d %H:%M:%S")
    }
    
    with open(out_file, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False)
        
    print("=" * 65)

    print("=" * 65)
    return out_file


def run_roi_tool(source_path: str = None, output_config: str = None):
    """Khởi chạy công cụ vẽ ROI tương tác bằng chuột cho 1 ảnh hoặc 1 video."""
    global all_polygons, current_polygon, CONFIG_FILE

    all_polygons.clear()
    current_polygon.clear()

    if source_path is None or not os.path.exists(source_path):
        video_dir = "data/videos"

        source_path = os.path.join(video_dir, files[0]) if files else "data/videos/sample_cctv.mp4"

    if output_config:
        CONFIG_FILE = output_config
    else:
        CONFIG_FILE = get_default_config_for_source(source_path)

    os.makedirs(os.path.dirname(CONFIG_FILE) or ".", exist_ok=True)

    # Nạp cấu hình ROI đã có trước đó nếu file tồn tại
    if os.path.exists(CONFIG_FILE):
        try:
            with open(CONFIG_FILE, "r", encoding="utf-8") as f:
                saved_data = json.load(f)

        except Exception:
            pass

    is_image = source_path.lower().endswith(('.jpg', '.jpeg', '.png', '.bmp', '.webp'))

    if is_image:
        base_frame = cv2.imread(source_path)
        if base_frame is None:
            print(f"[LỖI] Không thể đọc ảnh: {source_path}")
            return False
        height, width = base_frame.shape[:2]
    else:
        cap = cv2.VideoCapture(source_path)
        if not cap.isOpened():
            print(f"[LỖI] Không thể mở video: {source_path}")
            return False
        width = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH))
        height = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT))
        ret, base_frame = cap.read()
        cap.release()
        if not ret or base_frame is None:
            print("[LỖI] Không thể đọc frame từ video.")
            return False

    window_name = f"ROI Tool - {os.path.basename(source_path)} (Nhan 's' de luu, 'q' de thoat)"
    cv2.namedWindow(window_name, cv2.WINDOW_NORMAL)
    cv2.resizeWindow(window_name, min(width, 1050), min(height, 750))
    cv2.setMouseCallback(window_name, mouse_handler)

    print("=" * 70)
    print("CÔNG CỤ THIẾT LẬP ĐA VÙNG VỈA HÈ (MULTI-ROI TOOL)")
    print(f"  Nguồn dữ liệu : {source_path} ({width}x{height})")

    print(" 4. Nhấn phím 's'    : LƯU TẤT CẢ CÁC VÙNG VỈA HÈ VÀO FILE CONFIG JSON")
    print(" 5. Nhấn phím 'c'    : Xóa đa giác hiện tại để vẽ lại")
    print(" 6. Nhấn phím 'q'    : Thoát công cụ (hoặc chuyển sang ảnh tiếp theo)")
    print("-" * 70)

    has_saved = False

    while True:
        display_frame = draw_multi_roi_overlay(base_frame, all_polygons, current_polygon)

        if key == ord('n') or key == ord('N'):

        # Phím 's' hoặc 'S': Lưu cấu hình JSON
        elif key == ord('s') or key == ord('S'):
            to_save = list(all_polygons)
            if len(current_polygon) >= 3:
                to_save.append(list(current_polygon))

            if not to_save:
                print("[CẢNH BÁO] Chưa có vỉa hè nào có từ 3 điểm trở lên để lưu!")

                cam_name = os.path.splitext(os.path.basename(source_path))[0]

                print("[THÔNG BÁO] Đã lưu thành công. Bạn có thể nhấn 'q' để tiếp tục.")

        # Phím 'c' hoặc 'C': Xóa
        elif key == ord('c') or key == ord('C'):
            if current_polygon:
                current_polygon.clear()
                print("[THÔNG BÁO] Đã xóa điểm của vỉa hè đang vẽ dở.")
            elif all_polygons:
                all_polygons.clear()
                print("[THÔNG BÁO] Đã xóa toàn bộ tất cả các vỉa hè.")

        # Phím 'q' hoặc ESC: Thoát
        elif key == ord('q') or key == 27:
            # Tự động lưu nếu người dùng đã vẽ xong mà chưa bấm 's'
            if not has_saved and len(current_polygon) >= 3:
                all_polygons.append(list(current_polygon))
                cam_name = os.path.splitext(os.path.basename(source_path))[0]
                save_roi_config(all_polygons, width, height, camera_id=cam_name, target_path=CONFIG_FILE)
            break

    cv2.destroyAllWindows()
    return True

def get_default_roi_path_for_image(img_path: str) -> str:
    """Xác định đường dẫn file JSON ROI mặc định cho một tệp ảnh."""
    base_name = os.path.splitext(os.path.basename(img_path))[0]
    clean_name = base_name.replace(" ", "_").replace(",", "").replace("-", "_")
    candidates = [
        os.path.join("configs", f"roi_{clean_name}.json"),
        os.path.join("configs", f"roi_{base_name.lower().replace(' ', '_')}.json"),
    ]
    if "screenshot" in base_name.lower():
        candidates.append(os.path.join("configs", "roi_screenshot.json"))
        candidates.append(os.path.join("configs", "roi_Screenshot_2026_09_30_214856.json"))
    if "09_49" in base_name or "09-49" in base_name:
        candidates.append(os.path.join("configs", "roi_chatgpt_09_49.json"))
    if "09_51" in base_name or "09-51" in base_name:
        candidates.append(os.path.join("configs", "roi_chatgpt_09_51.json"))
        
    for c in candidates:
        if os.path.exists(c):
            return c
    return candidates[0]


def process_and_annotate(img, detections, checker=None, roi_manager=None, show_roi: bool = True):
    """
    Vẽ kết quả phát hiện và phán quyết vi phạm (nếu có ROI) lên ảnh.
    Trả về ảnh đã vẽ nhãn và danh sách chi tiết các biển.
    """
    vis = img.copy()
    h, w = img.shape[:2]
    
    # 1. Vẽ lớp phủ ROI vỉa hè nếu có cấu hình
    has_roi = False
    if roi_manager and len(roi_manager.polygons) > 0:
        has_roi = True
        if show_roi:
            vis = roi_manager.draw_overlay(vis, alpha=0.30)
        
    details = []
    
    for idx, det in enumerate(detections, 1):
        box = det["box"]
        conf = det["conf"]
        x1, y1, x2, y2 = box
        base_x = int((x1 + x2) / 2)
        base_y = int(y2)
        
        is_violation = False
        status_text = "BIEN QUANG CAO"
        overlap_pct = 0.0
        
        if checker and has_roi:
            # Kiểm tra vi phạm trên toàn bộ các đa giác vỉa hè (Multi-ROI)
            res = checker.check_multi(box, roi_manager.polygons)
            is_violation = res["is_violation"]
            overlap_pct = res["overlap_pct"]
            if is_violation:
                status_text = f"VI PHAM ({overlap_pct:.0f}%)"
            elif res.get("is_base_inside", False):
                status_text = f"MEP VIA HE ({overlap_pct:.0f}%)"
            else:
                status_text = f"HOP LE ({conf*100:.0f}%)"
        else:
            status_text = f"bienquangcao {conf*100:.0f}%"
            
        details.append({
            "idx": idx,
            "box": box,
            "conf": conf,
            "base_pt": (base_x, base_y),
            "is_violation": is_violation,
            "overlap_pct": overlap_pct,
            "status_text": status_text
        })
        
        # Vẽ box bằng hàm chuẩn trong utils/visualizer.py
        draw_detection(vis, box, status_text, is_violation=is_violation)
        
    return vis, details


def test_images_batch(
    img_dir: str = "data/images",
    model_path: str = "models/best.pt",
    conf_thresh: float = 0.25,
    imgsz: int = 1280,
    output_dir: str = "data/images/output_test_v8",
    roi_path: str = None
):
    """Chạy kiểm thử hàng loạt trên toàn bộ ảnh trong thư mục và lưu ảnh kết quả."""
    if not os.path.exists(img_dir):
        print(f"[LỖI] Thư mục không tồn tại: {img_dir}")
        return False

    os.makedirs(output_dir, exist_ok=True)
    images = [f for f in sorted(os.listdir(img_dir)) if f.lower().endswith(('.jpg', '.jpeg', '.png', '.webp'))]

    if not images:
        print(f"[CẢNH BÁO] Không tìm thấy ảnh nào trong thư mục: {img_dir}")
        return False

    print("=" * 75)
    print("      KIỂM THỬ MÔ HÌNH NHẬN DIỆN BIỂN QUẢNG CÁO (YOLOv8n v8)")
    print("=" * 75)
    print(f"[+] Trọng số mô hình : {model_path}")
    print(f"[+] Thư mục ảnh      : {os.path.abspath(img_dir)} ({len(images)} ảnh)")
    print(f"[+] Ngưỡng Conf / Size: conf={conf_thresh:.2f}, imgsz={imgsz}")
    print(f"[+] Thư mục lưu kết quả: {os.path.abspath(output_dir)}")
    
    checker = None
    default_roi_mgr = None
    if roi_path and os.path.exists(roi_path):
        print(f"[+] Nạp cấu hình ROI : {roi_path}")
        default_roi_mgr = ROIManager(roi_path)
        checker = ViolationChecker(threshold=0.30)
    print("-" * 75)

    detector = YOLODetector(model_path=model_path, conf_threshold=conf_thresh, imgsz=imgsz)
    
    total_detections = 0
    total_violations = 0
    total_time_ms = 0.0

    for idx, img_name in enumerate(images, start=1):
        img_path = os.path.join(img_dir, img_name)
        img = cv2.imread(img_path)
        if img is None:
            print(f"[!] Không thể đọc ảnh: {img_name}")
            continue

        h, w = img.shape[:2]
        
        # Tìm ROI riêng cho ảnh nếu không có roi_path cố định
        cur_roi_mgr = default_roi_mgr
        if cur_roi_mgr is None:
            custom_roi = get_default_roi_path_for_image(img_path)
            if os.path.exists(custom_roi):
                cur_roi_mgr = ROIManager(custom_roi)
                checker = ViolationChecker(threshold=0.30)

        # Đo thời gian suy luận
        t_start = time.perf_counter()
        detections = detector.detect(img)
        t_infer_ms = (time.perf_counter() - t_start) * 1000.0
        total_time_ms += t_infer_ms
        total_detections += len(detections)

        # Xử lý vẽ nhãn và ROI
        vis, details = process_and_annotate(img, detections, checker, cur_roi_mgr, show_roi=True)

        n_viol = sum(1 for d in details if d["is_violation"])
        total_violations += n_viol

        print(f"\n [ẢNH {idx}/{len(images)}] {img_name} ({w}x{h})")
        print(f"   ️  Thời gian suy luận: {t_infer_ms:.1f} ms | Số biển: {len(detections)} | Vi phạm vỉa hè: {n_viol}")
        
        for d in details:
            print(f"      • Biển #{d['idx']}: Box={d['box']} | Chân={d['base_pt']} | Conf={d['conf']*100:.1f}% | Nhãn: {d['status_text']}")

        # Lưu ảnh kết quả
        clean_name = f"v8_result_{idx}_{img_name.replace(' ', '_')}"
        out_path = os.path.join(output_dir, clean_name)
        cv2.imwrite(out_path, vis)
        print(f"    Đã lưu kết quả tại: {out_path}")

    # Báo cáo tổng kết
    avg_time = total_time_ms / len(images) if images else 0
    print("\n" + "=" * 75)
    print("                    TỔNG KẾT THỰC NGHIỆM")
    print("=" * 75)
    print(f"  • Tổng số ảnh đã kiểm thử    : {len(images)} ảnh")
    print(f"  • Tổng số biển phát hiện     : {total_detections} biển")
    if checker:
        print(f"  • Tổng số biển vi phạm vỉa hè: {total_violations} biển")
    print(f"  • Thời gian xử lý trung bình : {avg_time:.1f} ms/ảnh (~{1000.0/max(avg_time, 1):.1f} FPS)")
    print(f"  • Toàn bộ ảnh đã được lưu tại: {os.path.abspath(output_dir)}")
    print("=" * 75)
    return True


def test_single_image(
    image_path: str,
    model_path: str = "models/best.pt",
    conf_thresh: float = 0.25,
    imgsz: int = 1280,
    output_dir: str = "data/images/output_test_v8",
    roi_path: str = None,
    show_gui: bool = False
):
    """Kiểm thử một ảnh đơn lẻ."""
    if not os.path.exists(image_path):
        print(f"[LỖI] Không tìm thấy file ảnh: {image_path}")
        return False

    detector = YOLODetector(model_path=model_path, conf_threshold=conf_thresh, imgsz=imgsz)
    img = cv2.imread(image_path)
    if img is None:
        print(f"[LỖI] Không thể đọc ảnh: {image_path}")
        return False

    h, w = img.shape[:2]

    # Nạp ROI
    roi_mgr = None
    checker = None
    effective_roi = roi_path if roi_path and os.path.exists(roi_path) else get_default_roi_path_for_image(image_path)
    if os.path.exists(effective_roi):
        print(f"[+] Áp dụng ROI: {effective_roi}")
        roi_mgr = ROIManager(effective_roi)
        checker = ViolationChecker(threshold=0.30)

    t0 = time.perf_counter()
    detections = detector.detect(img)
    infer_ms = (time.perf_counter() - t0) * 1000.0

    print("=" * 70)
    print(f"[+] Ảnh kiểm thử : {image_path} ({w}x{h})")
    print(f"[+] Thời gian AI : {infer_ms:.1f} ms")
    print(f"[+] Phát hiện    : {len(detections)} biển hiệu!")
    
    vis, details = process_and_annotate(img, detections, checker, roi_mgr, show_roi=True)
    for d in details:
        print(f"   • Biển #{d['idx']}: Box={d['box']} | Conf={d['conf']*100:.1f}% | Nhãn: {d['status_text']}")

    os.makedirs(output_dir, exist_ok=True)
    out_path = os.path.join(output_dir, "result_" + os.path.basename(image_path).replace(' ', '_'))
    cv2.imwrite(out_path, vis)
    print(f"[OK] Đã lưu ảnh kết quả tại: {out_path}")
    print("=" * 70)

    if show_gui:
        try:
            window_name = f"Ket Qua AI - {os.path.basename(image_path)}"
            scale = min(1000.0 / w, 750.0 / h, 1.0)
            disp = cv2.resize(vis, (int(w * scale), int(h * scale)))
            print("Đang mở cửa sổ... Nhấn phím bất kỳ hoặc 'q' để đóng.")
            cv2.imshow(window_name, disp)
            cv2.waitKey(0)
            cv2.destroyAllWindows()
        except cv2.error as e:
            print(f"[!] Không thể mở cửa sổ hiển thị: {e}")
            print(f">> Ảnh kết quả đã được lưu đầy đủ tại: {out_path}")
    return True

"""
Module đo đạc FPS thời gian thực
"""
import time

class FPSTracker:
    def __init__(self, alpha: float = 0.15):
        self.alpha = alpha
        self.prev_time = None
        self.smoothed_fps = 0.0

    def update(self) -> float:
        curr_time = time.perf_counter()
        if self.prev_time is None:
            self.prev_time = curr_time
            return 0.0
            
        delta_t = curr_time - self.prev_time
        self.prev_time = curr_time
        
        if delta_t > 0:
            instant_fps = 1.0 / delta_t
            if self.smoothed_fps == 0.0:
                self.smoothed_fps = instant_fps
            else:
                self.smoothed_fps = self.alpha * instant_fps + (1.0 - self.alpha) * self.smoothed_fps
        return self.smoothed_fps

"""
Module hỗ trợ vẽ các thành phần đồ họa (Visualizer)
"""
import cv2
import numpy as np

def draw_fps_badge(frame, fps_value: float, video_fps: float = None):
    text = f"FPS: {fps_value:.1f}"
    if video_fps:
        text += f" | Native: {video_fps:.1f}"
        
    font = cv2.FONT_HERSHEY_SIMPLEX
    (tw, th), _ = cv2.getTextSize(text, font, 0.6, 2)
    cv2.rectangle(frame, (10, 10), (10 + tw + 10, 10 + th + 10), (30, 30, 30), -1)
    cv2.rectangle(frame, (10, 10), (10 + tw + 10, 10 + th + 10), (0, 255, 0), 1)
    cv2.putText(frame, text, (15, 10 + th + 3), font, 0.6, (0, 255, 0), 2, cv2.LINE_AA)

def draw_detection(frame, box, label, is_violation=False, track_id=None, status=None):
    x1, y1, x2, y2 = box
    
    # Xác định màu sắc theo 3 cấp độ: ĐỎ (Confirmed) | VÀNG CAM (Suspected) | XANH (Normal)
    if status == "CONFIRMED" or (status is None and is_violation):
        color = (0, 0, 255)       # Đỏ: Đã xác nhận vi phạm
    elif status == "SUSPECTED":
        color = (0, 165, 255)     # Vàng cam: Đang nghi vấn (chờ đủ N frames)
    else:
        color = (0, 255, 0)       # Xanh lá: Hợp lệ / Bình thường
    
    # Ghép ID vào nhãn nếu có
    display_text = f"ID:{track_id} | {label}" if (track_id is not None and track_id > 0) else label
    
    # 1. Bounding box
    cv2.rectangle(frame, (x1, y1), (x2, y2), color, 2)
    
    # 2. Tag nhãn
    font = cv2.FONT_HERSHEY_SIMPLEX
    (tw, th), _ = cv2.getTextSize(display_text, font, 0.5, 1)
    tag_top = max(0, y1 - th - 8)
    cv2.rectangle(frame, (x1, tag_top), (x1 + tw + 10, y1), color, -1)
    cv2.putText(frame, display_text, (x1 + 5, y1 - 4), font, 0.5, (255, 255, 255), 1, cv2.LINE_AA)
    
    # 3. Ba điểm chân tiếp đất (Chân trái, Giữa, Chân phải)
    pt_left = (int(x1), int(y2))
    pt_center = (int((x1 + x2) / 2), int(y2))
    pt_right = (int(x2), int(y2))
    
    # Thanh ngang chân đế
    cv2.line(frame, pt_left, pt_right, (0, 255, 255), 2, cv2.LINE_AA)
    
    # Vẽ 3 chấm tròn vàng viền đen đánh dấu 3 chân tiếp xúc
    for pt in [pt_left, pt_center, pt_right]:
        cv2.circle(frame, pt, 5, (0, 255, 255), -1)
        cv2.circle(frame, pt, 5, (0, 0, 0), 1)

Dựa trên thực nghiệm quét 9 giá trị ngưỡng từ 10% đến 60%, ngưỡng **30%** đạt điểm F1-Score cao nhất **(85.7%)**, cân bằng tối ưu giữa:
